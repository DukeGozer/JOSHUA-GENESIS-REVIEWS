using System.Diagnostics;
using System.IO;
using System.IO.Pipes;
using System.Text;
using System.Text.Json;
using System.Windows;
using System.Windows.Threading;
using System.Windows.Media;
using Genesis.Contracts;

namespace Genesis.UI;

public partial class MainWindow : Window
{
    private readonly DispatcherTimer _timer = new() { Interval = TimeSpan.FromSeconds(1) };
    private bool _refreshing;
    private sealed record JobRow(string Id, string Title, string State, int Completed, int Total)
    {
        public override string ToString() =>
            $"{Title}\n{State} – {Completed}/{Total}  ·  {(Id.Length > 8 ? Id[..8] : Id)}";
    }

    public MainWindow()
    {
        InitializeComponent();
        Loaded += async (_, _) => { await EnsureHostAsync(); await RefreshAsync(); await LoadModelsAsync(); _timer.Start(); };
        Closed += (_, _) => _timer.Stop(); // Closing the UI MUST NOT terminate the Host.
        _timer.Tick += async (_, _) => await RefreshAsync();
    }

    private static async Task<Response> SendAsync(Request request)
    {
        using var pipe = new NamedPipeClientStream(".", Protocol.ActivePipeName, PipeDirection.InOut,
            PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
        await pipe.ConnectAsync(1500);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        using var reader = new StreamReader(pipe, Encoding.UTF8, leaveOpen: true);
        await writer.WriteLineAsync(JsonSerializer.Serialize(request, Protocol.WireJson));
        using var timeout = new CancellationTokenSource(request.Command == "chat" ? TimeSpan.FromMinutes(3) : TimeSpan.FromSeconds(10));
        var reply = await ReadBoundedReplyAsync(reader, 4 * 1024 * 1024, timeout.Token);
        return reply is null ? new(false, "Keine Antwort vom Host")
            : JsonSerializer.Deserialize<Response>(reply, Protocol.WireJson)
              ?? new(false, "Ungültige Antwort");
    }

    private static async Task<string?> ReadBoundedReplyAsync(StreamReader reader, int limit, CancellationToken cancellation)
    {
        var result = new StringBuilder();
        var buffer = new char[1];
        while (true)
        {
            var count = await reader.ReadAsync(buffer.AsMemory(), cancellation);
            if (count == 0) return result.Length == 0 ? null : throw new InvalidDataException("Unvollständige Antwort.");
            if (buffer[0] == '\n') return result.ToString().TrimEnd('\r');
            if (result.Length >= limit) throw new InvalidDataException("Host-Antwort überschreitet das Größenlimit.");
            result.Append(buffer[0]);
        }
    }

    private static string? ReadControlToken()
    {
        var dataDirectory = (Protocol.TestOverridesEnabled
            ? Environment.GetEnvironmentVariable("GENESIS_DATA_DIR") : null)
            ?? Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "JoshuaSpike5Genesis");
        var path = Path.Combine(dataDirectory, "control.token");
        return File.Exists(path) ? File.ReadAllText(path).Trim() : null;
    }

    private static async Task EnsureHostAsync()
    {
        try { if ((await SendAsync(new Request("status"))).Success) return; } catch { }
        var path = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "..", "Host", "Genesis.Host.exe"));
        if (!File.Exists(path)) return;
        using var process = Process.Start(new ProcessStartInfo(path)
        {
            UseShellExecute = false, CreateNoWindow = true
        });
        for (var i = 0; i < 15; i++)
        {
            await Task.Delay(200);
            try { if ((await SendAsync(new Request("status"))).Success) return; } catch { }
            if (process?.HasExited == true) return;
        }
    }

    private async Task RefreshAsync()
    {
        if (_refreshing) return;
        _refreshing = true;
        try
        {
            var response = await SendAsync(new Request("status"));
            ConnectionText.Text = response.Success ? "● " + response.Message : response.Message;
            if (!response.Success) ClearStateViews();
            if (response.Data is { } data)
            {
                DecisionsList.ItemsSource = data.Decisions.OrderByDescending(x => x.CreatedAt)
                    .Select(x => $"{x.CreatedAt.LocalDateTime:g}  {x.Text}").ToArray();
                var selected = (JobsList.SelectedItem as JobRow)?.Id;
                JobsList.ItemsSource = data.Jobs.OrderByDescending(x => x.CreatedAt)
                    .Select(x => new JobRow(x.Id, x.Title, x.State, x.Completed, x.Total)).ToArray();
                JobsList.SelectedItem = JobsList.Items.Cast<JobRow>().FirstOrDefault(x => x.Id == selected);
            }
        }
        catch
        {
            ConnectionText.Text = "○ Host nicht erreichbar – Log prüfen";
            ClearStateViews();
        }
        finally { _refreshing = false; }
    }

    private void ClearStateViews()
    {
        DecisionsList.ItemsSource = null;
        JobsList.ItemsSource = null;
    }

    private void ApplyTheme(bool dark)
    {
        var palette = dark
            ? new Dictionary<string, string>
            {
                ["WindowBg"] = "#161E2A", ["PanelBg"] = "#202C3A",
                ["PanelBorder"] = "#364858", ["MainText"] = "#E4F0F5",
                ["SubText"] = "#B3CAD4", ["Accent"] = "#66C1C9",
                ["ButtonBg"] = "#2B4050", ["InputBg"] = "#162532"
            }
            : new Dictionary<string, string>
            {
                ["WindowBg"] = "#E5EBEF", ["PanelBg"] = "#EEF3F6",
                ["PanelBorder"] = "#CDD9DF", ["MainText"] = "#1B3646",
                ["SubText"] = "#516C7B", ["Accent"] = "#17899D",
                ["ButtonBg"] = "#E4EDF1", ["InputBg"] = "#FAFCFD"
            };
        foreach (var item in palette)
        {
            Resources[item.Key] = new SolidColorBrush((Color)ColorConverter.ConvertFromString(item.Value));
        }
    }

    private void LightTheme_Click(object sender, RoutedEventArgs e) => ApplyTheme(false);
    private void DarkTheme_Click(object sender, RoutedEventArgs e) => ApplyTheme(true);

    private async Task LoadModelsAsync()
    {
        try
        {
            var response = await SendAsync(new Request("models"));
            if (!response.Success)
            {
                ChatOutput.Text = "Ollama: " + response.Message;
                return;
            }
            var selected = ModelPicker.SelectedItem as string;
            ModelPicker.ItemsSource = response.Models ?? [];
            if (selected is not null && ModelPicker.Items.Contains(selected))
                ModelPicker.SelectedItem = selected;
            else if (ModelPicker.Items.Count > 0)
                ModelPicker.SelectedIndex = 0;
            ChatOutput.Text = response.Models?.Length > 0
                ? "Ollama verbunden. Modell auswählen, Nachricht eingeben und senden."
                : "Ollama läuft, aber es wurde noch kein Modell gefunden.";
        }
        catch (Exception ex) { ChatOutput.Text = "Ollama nicht erreichbar: " + ex.Message; }
    }

    private async void LoadModels_Click(object sender, RoutedEventArgs e) => await LoadModelsAsync();

    private async void SendChat_Click(object sender, RoutedEventArgs e)
    {
        if (ModelPicker.SelectedItem is not string model)
        {
            ChatOutput.Text = "Bitte zuerst ein Ollama-Modell auswählen.";
            return;
        }
        if (string.IsNullOrWhiteSpace(ChatInput.Text)) return;
        var prompt = ChatInput.Text;
        SendChatButton.IsEnabled = false;
        ChatOutput.Text = "JOSHUA arbeitet mit " + model + "...";
        try
        {
            var answer = await SendAsync(new Request("chat", Text: prompt, Model: model));
            ChatOutput.Text = answer.Success ? (answer.ChatText ?? "(Leere Antwort)") : answer.Message;
            if (answer.Success) ChatInput.Clear();
        }
        catch (Exception ex) { ChatOutput.Text = "Chat fehlgeschlagen: " + ex.Message; }
        finally { SendChatButton.IsEnabled = true; }
    }

    private async void StartHost_Click(object sender, RoutedEventArgs e)
    {
        try { await EnsureHostAsync(); await RefreshAsync(); }
        catch (Exception ex) { MessageBox.Show(ex.Message, "Host-Start"); }
    }

    private async void StopHost_Click(object sender, RoutedEventArgs e)
    {
        try
        {
            var result = await SendAsync(new Request("stop", AuthToken: ReadControlToken()));
            if (!result.Success) MessageBox.Show(result.Message);
            await Task.Delay(300);
            ClearStateViews();
            await RefreshAsync();
        }
        catch (Exception ex) { MessageBox.Show(ex.Message, "Host beenden"); }
    }

    private async void CancelSelected_Click(object sender, RoutedEventArgs e)
    {
        if (JobsList.SelectedItem is not JobRow selected)
        {
            MessageBox.Show("Bitte zuerst einen Auftrag auswählen.");
            return;
        }
        try
        {
            var result = await SendAsync(new Request("cancel", Id: selected.Id));
            if (!result.Success) MessageBox.Show(result.Message);
            await RefreshAsync();
        }
        catch (Exception ex) { MessageBox.Show(ex.Message, "Auftrag abbrechen"); }
    }

    private async void StartJob_Click(object sender, RoutedEventArgs e)
    {
        try
        {
            var result = await SendAsync(new Request("start-demo"));
            if (!result.Success) MessageBox.Show(result.Message);
            await RefreshAsync();
        }
        catch (Exception ex) { MessageBox.Show(ex.Message, "Verbindungsfehler"); }
    }

    private async void Remember_Click(object sender, RoutedEventArgs e)
    {
        if (string.IsNullOrWhiteSpace(DecisionInput.Text)) return;
        try
        {
            var result = await SendAsync(new Request("remember", DecisionInput.Text));
            if (result.Success) DecisionInput.Clear(); else MessageBox.Show(result.Message);
            await RefreshAsync();
        }
        catch (Exception ex) { MessageBox.Show(ex.Message, "Verbindungsfehler"); }
    }

    private async void Refresh_Click(object sender, RoutedEventArgs e) => await RefreshAsync();
}

using System.Globalization;
using Genesis.Contracts;
using Microsoft.Data.Sqlite;

namespace Genesis.Storage;

/// <summary>
/// SQLite is the authoritative GENESIS 0.2 state. All decisions/jobs are committed
/// in a single transaction; interrupted transactions never expose partial snapshots.
/// </summary>
public sealed class SnapshotStore : IDisposable
{
    private readonly SqliteConnection _connection;

    public SnapshotStore(string dataDirectory)
    {
        Directory.CreateDirectory(dataDirectory);
        var path = Path.Combine(dataDirectory, "genesis.db");
        _connection = new SqliteConnection(new SqliteConnectionStringBuilder
        {
            DataSource = path,
            Mode = SqliteOpenMode.ReadWriteCreate,
            Cache = SqliteCacheMode.Private,
            Pooling = false
        }.ToString());
        _connection.Open();
        Exec("PRAGMA journal_mode=WAL;");
        Exec("PRAGMA busy_timeout=5000;");
        Exec("PRAGMA synchronous=FULL;");
        Exec("CREATE TABLE IF NOT EXISTS metadata (key TEXT PRIMARY KEY, value TEXT NOT NULL);");
        Exec("CREATE TABLE IF NOT EXISTS decisions (id TEXT PRIMARY KEY, text TEXT NOT NULL, created_at TEXT NOT NULL);");
        Exec("CREATE TABLE IF NOT EXISTS jobs (id TEXT PRIMARY KEY, title TEXT NOT NULL, state TEXT NOT NULL, completed INTEGER NOT NULL, total INTEGER NOT NULL, created_at TEXT NOT NULL, error TEXT);");
        var version = GetMeta("schema_version");
        if (version is not null && version != "1")
            throw new InvalidDataException("Unsupported SQLite schema " + version);
    }

    private void Exec(string sql)
    {
        using var c = _connection.CreateCommand();
        c.CommandText = sql;
        c.ExecuteNonQuery();
    }

    private string? GetMeta(string key)
    {
        using var c = _connection.CreateCommand();
        c.CommandText = "SELECT value FROM metadata WHERE key=$key";
        c.Parameters.AddWithValue("$key", key);
        return c.ExecuteScalar() as string;
    }

    public Snapshot? Read()
    {
        if (GetMeta("initialized") != "1") return null;
        var s = new Snapshot { SchemaVersion = 1 };
        var updated = GetMeta("updated_at");
        if (updated is not null) s.UpdatedAt = DateTimeOffset.Parse(updated, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind);
        using (var c = _connection.CreateCommand())
        {
            c.CommandText = "SELECT id,text,created_at FROM decisions ORDER BY created_at,id";
            using var r = c.ExecuteReader();
            while (r.Read())
                s.Decisions.Add(new Decision(r.GetString(0), r.GetString(1),
                    DateTimeOffset.Parse(r.GetString(2), CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind)));
        }
        using (var c = _connection.CreateCommand())
        {
            c.CommandText = "SELECT id,title,state,completed,total,created_at,error FROM jobs ORDER BY created_at,id";
            using var r = c.ExecuteReader();
            while (r.Read())
                s.Jobs.Add(new Job(r.GetString(0), r.GetString(1), r.GetString(2),
                    r.GetInt32(3), r.GetInt32(4),
                    DateTimeOffset.Parse(r.GetString(5), CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind),
                    r.IsDBNull(6) ? null : r.GetString(6)));
        }
        return s;
    }

    public void Commit(Snapshot candidate)
    {
        if (candidate.SchemaVersion != 1) throw new InvalidDataException("Unsupported Snapshot version.");
        using var tx = _connection.BeginTransaction();
        using (var clean = _connection.CreateCommand())
        {
            clean.Transaction = tx;
            clean.CommandText = "DELETE FROM decisions; DELETE FROM jobs;";
            clean.ExecuteNonQuery();
        }
        foreach (var d in candidate.Decisions)
        {
            using var c = _connection.CreateCommand();
            c.Transaction = tx;
            c.CommandText = "INSERT INTO decisions (id,text,created_at) VALUES ($id,$text,$date)";
            c.Parameters.AddWithValue("$id", d.Id);
            c.Parameters.AddWithValue("$text", d.Text);
            c.Parameters.AddWithValue("$date", d.CreatedAt.ToString("O", CultureInfo.InvariantCulture));
            c.ExecuteNonQuery();
        }
        foreach (var j in candidate.Jobs)
        {
            using var c = _connection.CreateCommand();
            c.Transaction = tx;
            c.CommandText = @"INSERT INTO jobs (id,title,state,completed,total,created_at,error)
                VALUES ($id,$title,$state,$completed,$total,$date,$error)";
            c.Parameters.AddWithValue("$id", j.Id);
            c.Parameters.AddWithValue("$title", j.Title);
            c.Parameters.AddWithValue("$state", j.State);
            c.Parameters.AddWithValue("$completed", j.Completed);
            c.Parameters.AddWithValue("$total", j.Total);
            c.Parameters.AddWithValue("$date", j.CreatedAt.ToString("O", CultureInfo.InvariantCulture));
            c.Parameters.AddWithValue("$error", (object?)j.Error ?? DBNull.Value);
            c.ExecuteNonQuery();
        }
        foreach (var (key,value) in new[] { ("schema_version", "1"), ("updated_at", candidate.UpdatedAt.ToString("O", CultureInfo.InvariantCulture)), ("initialized", "1") })

            c.ExecuteNonQuery();
        }
        tx.Commit();
