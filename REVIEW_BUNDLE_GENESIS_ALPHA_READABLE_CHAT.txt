                        server = null;

                        activeClients.Add(task);
                        activeClients.RemoveAll(t => t.IsCompleted);
                    }
                    catch (OperationCanceledException) when (StopSignal.IsCancellationRequested)
                    {
                        server?.Dispose();
                        clientSlots.Release();

                }

    private static async Task ProcessClientAsync(
        NamedPipeServerStream pipe, CancellationToken stopping, SemaphoreSlim slots)
    {
        try
        {
            using (pipe) await HandleClient(pipe, stopping);
        }
        catch (Exception ex) { Log("Unhandled client error: " + ex); }
        finally { slots.Release(); }
    }

    private static async Task HandleClient(NamedPipeServerStream pipe, CancellationToken stopping)
    {
        // Handle one request per connection; every read has a deadline and size bound.
        using var reader = new StreamReader(pipe, Encoding.UTF8, detectEncodingFromByteOrderMarks: false, leaveOpen: true);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        try
        {
            using var timeout = CancellationTokenSource.CreateLinkedTokenSource(stopping);
            timeout.CancelAfter(TimeSpan.FromSeconds(10));
            var line = await PipeLineCodec.ReadBoundedAsync(reader, MaxRequestChars, timeout.Token);
            if (line is null) return;
            Request? request;
            try { request = JsonSerializer.Deserialize<Request>(line, Protocol.WireJson); }
            catch (JsonException)
            {
                await writer.WriteLineAsync(JsonSerializer.Serialize(
                    new Response(false, "Ungültiges JSON."), Protocol.WireJson));
                return;
            }

            // Reject unauthorized clients before model discovery or state access.
            // This is defense against accidental local callers; same-user malware can still read the token.
            if (!IsAuthorized(request))
            {
                await writer.WriteLineAsync(JsonSerializer.Serialize(
                    new Response(false, "Host-Zugriff nicht autorisiert."), Protocol.WireJson));
                return;
            }

            // Only model discovery waits on its connection; chat is now a background job.
            if (request?.Command?.Equals("models", StringComparison.OrdinalIgnoreCase) == true)
            {
                if (!ModelListGate.Wait(0))
                {
                    await writer.WriteLineAsync(JsonSerializer.Serialize(
                        new Response(false, "Modellabfrage ausgelastet."), Protocol.WireJson));
                    return;
                }
                try
                {
                    using var modelTimeout = CancellationTokenSource.CreateLinkedTokenSource(stopping);
                    modelTimeout.CancelAfter(TimeSpan.FromSeconds(10));
                    var modelResponse = await HandleModelRequestAsync(request, modelTimeout.Token);
                    await writer.WriteLineAsync(JsonSerializer.Serialize(modelResponse, Protocol.WireJson));
                }
                finally { ModelListGate.Release(); }
                return;
            }

            Response response;
            lock (StateGate)
            {
                try { response = Execute(request); }
                catch (IOException ex) { Log("Persistence I/O error: " + ex); response = new(false, "Speichern fehlgeschlagen. Keine Änderung bestätigt."); }
                catch (UnauthorizedAccessException ex) { Log("Persistence permission error: " + ex); response = new(false, "Speichern nicht erlaubt."); }
            }
            var shouldStop = request?.Command?.Equals("stop", StringComparison.OrdinalIgnoreCase) == true && response.Success;
            try
            {
                await writer.WriteLineAsync(JsonSerializer.Serialize(PrepareWireResponse(response), Protocol.WireJson));
            }
            finally
            {
                // A disconnected client must not leave GENESIS running but refusing new work.
                if (shouldStop) StopSignal.Cancel();
            }
        }
        catch (OperationCanceledException) when (!stopping.IsCancellationRequested)
        {
            await TryReplyAsync(writer, new(false, "Zeitüberschreitung beim Lesen."));
        }
        catch (InvalidDataException) { await TryReplyAsync(writer, new(false, "Anfrage zu groß.")); }
        catch (OperationCanceledException) when (stopping.IsCancellationRequested) { }
        catch (Exception ex) { Log("Client error: " + ex); await TryReplyAsync(writer, new(false, "Anfrage fehlgeschlagen.")); }
    }

    private static void SetChatState(string id, string state, string? result = null, string? error = null)
    {
        lock (StateGate)
        {
            var jobIndex = _state.Jobs.FindIndex(j => j.Id == id);
            if (jobIndex < 0 || _state.Jobs[jobIndex].State is "canceled" or "done" or "failed") return;
            if (state == "running" && _state.Jobs[jobIndex].State == "canceling") return;
            var updated = DateTimeOffset.UtcNow;
            (_store ?? throw new InvalidOperationException("SQLite unavailable."))
                .UpdateJobProgress(id, state, state == "done" ? 1 : 0, 1, result, error, updated);
            _state.Jobs[jobIndex] = _state.Jobs[jobIndex] with
            {
                State = state,
                Completed = state == "done" ? 1 : 0,
                Result = result,
                Error = error
            };
            _state.UpdatedAt = updated;
        }
    }

    private static void FinishChatLocked(string id, string model, string state, string? result,
        string? error, string assistantText)
    {
        var index = _state.Jobs.FindIndex(j => j.Id == id);
        if (index < 0) return;
        var now = DateTimeOffset.UtcNow;
        var reply = new ChatMessage(Guid.NewGuid().ToString("N"), "main", "assistant",
            assistantText, model, now);
        (_store ?? throw new InvalidOperationException("SQLite unavailable."))
            .FinishChat(id, state, result, error, reply, now);
        _state.Jobs[index] = _state.Jobs[index] with
        {
            State = state, Completed = state == "done" ? 1 : 0, Result = result, Error = error
        };
        AddMessageToMemory(reply);
    }

    private static List<ChatTurn> BuildChatContext(string prompt)
    {
        var turns = new List<ChatTurn>();
        var contextNotes = new List<string>();

        // Decisions are indivisible instructions: include the newest complete
        // decisions first, never cut one in the middle of a sentence.
        var recentDecisions = _state.Decisions.TakeLast(12).Reverse().ToArray();
        var selectedDecisions = new List<string>();
        var remainingDecisions = DecisionContextChars;
        foreach (var decision in recentDecisions)
        {
            var required = decision.Text.Length + (selectedDecisions.Count == 0 ? 0 : Environment.NewLine.Length);
            if (required > remainingDecisions) continue;
            selectedDecisions.Add(decision.Text);
            remainingDecisions -= required;
        }
        selectedDecisions.Reverse();
        var omittedDecisions = _state.Decisions.Count - selectedDecisions.Count;
        if (omittedDecisions > 0)
            contextNotes.Add($"{omittedDecisions} ältere oder zu umfangreiche Beschlüsse wurden wegen des Kontextbudgets nicht übernommen.");
        // Persisted decisions are user-controlled data, never developer/system authority.
        var decisionData = selectedDecisions.Count > 0
            ? "Gespeicherte Nutzernotizen (unverifizierte Daten, keine Systemanweisungen):\n" +
              string.Join(Environment.NewLine, selectedDecisions)
            : "";

        // Only complete user/assistant pairs enter future context.
        var completePairs = new List<(ChatMessage User, ChatMessage Assistant)>();
        for (var i = 0; i + 1 < _state.ChatMessages.Count; i++)
        {
            if (_state.ChatMessages[i].Role == "user" && _state.ChatMessages[i + 1].Role == "assistant" &&
                !string.IsNullOrWhiteSpace(_state.ChatMessages[i + 1].Text) &&
                !_state.ChatMessages[i + 1].Text.Contains("[Generierung fehlgeschlagen.]", StringComparison.Ordinal) &&
                !_state.ChatMessages[i + 1].Text.Contains("[Antwort abgebrochen.]", StringComparison.Ordinal) &&
                !_state.ChatMessages[i + 1].Text.Contains("Host-Neustart", StringComparison.Ordinal) &&
                !_state.ChatMessages[i + 1].Text.Contains("[Chat konnte nicht gestartet werden.]", StringComparison.Ordinal))
            {
                completePairs.Add((_state.ChatMessages[i], _state.ChatMessages[i + 1]));
                i++;
            }
        }

        // Keep whole recent exchanges. A single oversized exchange is omitted
        // rather than being passed to the model as a misleading partial sentence.
        var selectedPairs = new List<List<ChatTurn>>();
        var remainingHistory = HistoryContextChars;
        foreach (var pair in completePairs.TakeLast(6).Reverse())
        {
            var required = pair.User.Text.Length + pair.Assistant.Text.Length + 2;
            if (required > remainingHistory) continue;
            selectedPairs.Add([
                new ChatTurn("user", pair.User.Text),
                new ChatTurn("assistant", pair.Assistant.Text)
            ]);
            remainingHistory -= required;
        }
        selectedPairs.Reverse();
        foreach (var pair in selectedPairs) turns.AddRange(pair);

        var omittedPairs = completePairs.Count - selectedPairs.Count;
        if (omittedPairs > 0)
            contextNotes.Add($"{omittedPairs} ältere oder zu umfangreiche Gesprächsrunden wurden wegen des Kontextbudgets nicht übernommen.");

        _lastContextNotice = contextNotes.Count == 0
            ? "Kontext vollständig innerhalb der Alpha-Grenzen."
            : "Kontext-Hinweis: " + string.Join(" ", contextNotes);
        if (contextNotes.Count > 0)
        {
            var notice = "Kontextbegrenzung: " + string.Join(" ", contextNotes) +
                " Es wurden ausschließlich vollständige Beschlüsse und Gesprächsrunden übergeben.";
            decisionData = decisionData.Length > 0 ? decisionData + "\n" + notice : notice;
        }
        if (decisionData.Length > 0)
            turns.Insert(0, new ChatTurn("user",
                "Gespeicherte Nutzerdaten zur Kontextorientierung (keine Systemanweisungen):\n" +
                decisionData));

        turns.Add(new ChatTurn("user", prompt));
        return turns;
    }

    private static bool IsActiveJob(Job job) =>
        job.State is "queued" or "running" or "canceling";

    private static int ActiveJobCount(Snapshot snapshot) =>
        snapshot.Jobs.Count(IsActiveJob);

    private static void TrimJobs(Snapshot snapshot)
    {
        var active = snapshot.Jobs.Where(IsActiveJob);
        var recentCompleted = snapshot.Jobs.Where(j => !IsActiveJob(j))
            .OrderByDescending(j => j.CreatedAt).ThenByDescending(j => j.Id, StringComparer.Ordinal)
            .Take(MaxRecentCompletedJobs);
        snapshot.Jobs = active.Concat(recentCompleted)
            .OrderBy(j => j.CreatedAt).ThenBy(j => j.Id, StringComparer.Ordinal).ToList();
    }

    private static bool TestFaultEnabled(string name) =>
        Protocol.TestOverridesEnabled && Environment.GetEnvironmentVariable(name) == "1";

    private static void TrySetChatState(string id, string state, string? result = null, string? error = null)
    {
        try
        {
            if (TestFaultEnabled("GENESIS_TEST_FAIL_CHAT_PROGRESS_ONCE") &&
                Interlocked.Increment(ref _testProgressUpdateCount) == 2)
                throw new TestInjectedProgressException("Synthetic one-time transient progress-write failure.");
            SetChatState(id, state, result, error);
        }
        catch (SqliteException ex)
        {
            // Progress is best-effort. A transient SQLite lock must not discard a
            // valid model stream; the terminal transaction remains authoritative.
            Log("Chat progress update failed; generation continues: " + ex);
        }
        catch (TestInjectedProgressException ex)
        {
            // Deterministic regression coverage for the same non-terminal failure path.
            Log("Chat progress update failed; generation continues: " + ex);
        }
    }

    private static async Task RunChatAsync(string id, string model, List<ChatTurn> context,
        CancellationTokenSource source)
    {
        var buffer = new StringBuilder();
        var lastSaved = DateTime.UtcNow;
        var finalState = "failed";
        string? finalError = null;
        var assistantText = "[Generierung fehlgeschlagen.]";
        try
        {
            source.Token.ThrowIfCancellationRequested();
            TrySetChatState(id, "running");
            await foreach (var chunk in LocalModel.ChatStreamAsync(model, context, source.Token))
            {
                buffer.Append(chunk);
                if ((DateTime.UtcNow - lastSaved).TotalMilliseconds >= 400)
                {
                    TrySetChatState(id, "running", buffer.ToString());
                    lastSaved = DateTime.UtcNow;
                }
            }
            source.Token.ThrowIfCancellationRequested();
            if (string.IsNullOrWhiteSpace(buffer.ToString()))
                throw new InvalidDataException("Ollama lieferte keine sichtbare Modellantwort.");
            finalState = "done";
            assistantText = buffer.ToString();
        }
        catch (OperationCanceledException)
        {
            finalState = "canceled";
            finalError = "Generierung abgebrochen.";
            assistantText = buffer.Length == 0 ? "[Antwort abgebrochen.]" :
                buffer + Environment.NewLine + Environment.NewLine + "[Antwort abgebrochen.]";
        }
        catch (Exception ex)
        {
            finalState = "failed";
            finalError = ex is HttpRequestException ? "Ollama: " + ex.Message :
                "Generierung fehlgeschlagen: " + ex.Message;
            Log("Chat worker error: " + ex);
            assistantText = buffer.Length == 0 ? "[Generierung fehlgeschlagen.]" :
                buffer + Environment.NewLine + Environment.NewLine + "[Generierung fehlgeschlagen.]";
        }
        finally
        {
            // Test-only delay makes the canceling state observable while a fake model
            // simulates a slow network teardown. It is ignored in normal operation.
            if (source.IsCancellationRequested && Protocol.TestOverridesEnabled &&
                int.TryParse(Environment.GetEnvironmentVariable("GENESIS_TEST_CANCEL_RELEASE_DELAY_MS"), out var delayMs))
                await Task.Delay(Math.Clamp(delayMs, 0, 5000));

            lock (StateGate)
            {
                // The visible terminal state changes only after the model gate is free.
                // Holding StateGate across release + persistence prevents a new request
                // or cancel command from observing an impossible intermediate state.
                if (source.IsCancellationRequested && finalState == "done")
                {
                    finalState = "canceled";
                    finalError = "Generierung abgebrochen.";
                    assistantText = buffer.Length == 0 ? "[Antwort abgebrochen.]" :
                        buffer + Environment.NewLine + Environment.NewLine + "[Antwort abgebrochen.]";
                }
                ModelGate.Release();
                try
                {
                    if (TestFaultEnabled("GENESIS_TEST_FAIL_CHAT_FINISH"))
                        throw new IOException("Synthetic test-only terminal persistence failure.");
                    FinishChatLocked(id, model, finalState,
                        buffer.ToString(),
                        finalError, assistantText);
                }
                catch (Exception writeError)
                {
                    Log("Could not persist chat terminal state: " + writeError);
                    try
                    {
                        if (TestFaultEnabled("GENESIS_TEST_FAIL_CHAT_FALLBACK"))
                            throw new IOException("Synthetic test-only fallback persistence failure.");
                        var now = DateTimeOffset.UtcNow;
                        (_store ?? throw new InvalidOperationException("SQLite unavailable."))
                            .UpdateJobProgress(id, "failed", 0, 1, buffer.ToString(),
                                "Persistenzfehler beim Abschluss: " + writeError.Message, now);
                        var index = _state.Jobs.FindIndex(j => j.Id == id);
                        if (index >= 0)
                            _state.Jobs[index] = _state.Jobs[index] with
                            { State = "failed", Result = buffer.ToString(), Error = "Persistenzfehler beim Abschluss." };
                        _state.UpdatedAt = now;
                    }
                    catch (Exception fallbackError)
                    {
                        Log("Could not persist fallback failure: " + fallbackError);
                        var failedIndex = _state.Jobs.FindIndex(j => j.Id == id);
                        if (failedIndex >= 0)
                            _state.Jobs[failedIndex] = _state.Jobs[failedIndex] with
                            {
                                State = "failed",
                                Result = buffer.ToString(),
                                Error = "Persistenzfehler: Abschluss konnte nicht in SQLite gespeichert werden."
                            };
                        _state.UpdatedAt = DateTimeOffset.UtcNow;
                        _persistenceFaulted = true;
                        _startupNotice = "WARNUNG: SQLite-Persistenz gestört. Host ist bis zum Neustart schreibgeschützt.";
                    }
                }
                ChatCancels.Remove(id);
            }
            source.Dispose();
        }
    }

    private static async Task TryReplyAsync(StreamWriter writer, Response response)
    {
        try { await writer.WriteLineAsync(JsonSerializer.Serialize(response, Protocol.WireJson)); }
        catch (IOException) { }
        catch (ObjectDisposedException) { }
    }

    private static Response PrepareWireResponse(Response response)
    {
        if (response.Data is null) return response;
        var source = response.Data;
        var latestChatId = source.Jobs.LastOrDefault(j => j.Kind == "chat")?.Id;
        var wire = new Snapshot
        {
            SchemaVersion = source.SchemaVersion,
            UpdatedAt = source.UpdatedAt,
            Decisions = source.Decisions.ToList(),
            Jobs = source.Jobs.Select(j => j with
            {
                Prompt = null,
                Result = (IsActiveJob(j) || j.Id == latestChatId) && j.Result is not null
                    ? j.Result[..Math.Min(j.Result.Length, 12000)] : null
            }).ToList(),
            ChatMessages = source.ChatMessages.TakeLast(40).Select(m => m with
            {
                Text = m.Text.Length <= 12000 ? m.Text : m.Text[..12000] +
                    "\n[Anzeige gekürzt; vollständige Nachricht bleibt in SQLite.]"
            }).ToList()
        };
        return response with { Data = wire };
    }

    private static bool IsAuthorized(Request? request)
    {
        var candidate = request?.AuthToken;
        return candidate is not null && candidate.Length == _controlToken.Length &&
            System.Security.Cryptography.CryptographicOperations.FixedTimeEquals(
                Encoding.ASCII.GetBytes(candidate), Encoding.ASCII.GetBytes(_controlToken));
    }

    private static Response Execute(Request? request)
    {
        if (string.IsNullOrWhiteSpace(request?.Command)) return new(false, "Befehl fehlt.");
        var command = request.Command.Trim().ToLowerInvariant();
        if (_persistenceFaulted && command is not ("status" or "stop"))
            return new(false, "SQLite-Persistenz ist gestört. Schreibende Aktionen sind bis zum kontrollierten Host-Neustart gesperrt.");
        if (!_acceptingWork && command is not ("status" or "stop"))
            return new(false, "GENESIS wird beendet und nimmt keine neuen Aufträge mehr an.");
        switch (command)
        {
            case "status":
                var statusMessage = _persistenceFaulted ? _startupNotice :
                    string.IsNullOrWhiteSpace(_lastContextNotice) ? _startupNotice : _startupNotice + " · " + _lastContextNotice;
                return new(true, statusMessage, Clone());
            case "remember":
                if (string.IsNullOrWhiteSpace(request.Text)) return new(false, "Bitte Beschlusstext angeben.");
                if (request.Text.Length > 2000) return new(false, "Beschlusstext zu lang (max. 2000 Zeichen).");
                if (_state.Decisions.Count >= MaxStoredDecisions) return new(false, "Beschlusslimit erreicht (500).");
                var decided = Clone();
                decided.Decisions.Add(new Decision(Guid.NewGuid().ToString("N"), request.Text.Trim(), DateTimeOffset.UtcNow));
                Save(decided);
                return new(true, "Beschluss gespeichert.", Clone());
            case "chat":
                if (string.IsNullOrWhiteSpace(request.Text) || request.Text.Length > MaxChatPromptChars ||
                    string.IsNullOrWhiteSpace(request.Model) || request.Model.Length > 160)
                    return new(false, $"Modell und Nachricht erforderlich (maximal {MaxChatPromptChars:N0} Zeichen).");
                if (ActiveJobCount(_state) >= MaxActiveJobs)
                    return new(false, $"Auftragslimit erreicht ({MaxActiveJobs} gleichzeitig aktive Aufträge).");
                if (!ModelGate.Wait(0))
                    return new(false, "Modell ausgelastet. Laufende Generierung zuerst abschließen oder abbrechen.");
                var id = Guid.NewGuid().ToString("N");
                var token = CancellationTokenSource.CreateLinkedTokenSource(StopSignal.Token);
                token.CancelAfter(TimeSpan.FromMinutes(3));
                var persisted = false;
                try
                {
                    var now = DateTimeOffset.UtcNow;
                    var ctx = BuildChatContext(request.Text);
                    var job = new Job(id, "Ollama: " + request.Model, "queued", 0, 1, now,
                        Kind: "chat", Model: request.Model, Prompt: request.Text);
                    var userMessage = new ChatMessage(Guid.NewGuid().ToString("N"), "main", "user",
                        request.Text, request.Model, now);
                    (_store ?? throw new InvalidOperationException("SQLite unavailable."))
                        .CreateChatRequest(job, userMessage);
                    persisted = true;
                    _state.Jobs.Add(job);
                    TrimJobs(_state);
                    AddMessageToMemory(userMessage);
                    ChatCancels.Add(id, token);
                    var task = Task.Run(() => RunChatAsync(id, request.Model!, ctx, token));
                    ChatTasks.Add(task);
                    ChatTasks.RemoveAll(t => t.IsCompleted);
                    return new(true, "Chat-Auftrag angenommen.", Clone());
                }
                catch (Exception ex)
                {
                    if (persisted)
                    {
                        try
                        {
                            var now = DateTimeOffset.UtcNow;
                            var marker = new ChatMessage(Guid.NewGuid().ToString("N"), "main", "assistant",
                                "[Chat konnte nicht gestartet werden.]", request.Model, now);
                            (_store ?? throw new InvalidOperationException("SQLite unavailable."))
                                .FinishChat(id, "failed", null, "Chat konnte nicht gestartet werden: " + ex.Message, marker, now);
                            var failedJobIndex = _state.Jobs.FindIndex(j => j.Id == id);
                            if (failedJobIndex >= 0) _state.Jobs[failedJobIndex] = _state.Jobs[failedJobIndex] with
                            { State = "failed", Error = "Chat konnte nicht gestartet werden: " + ex.Message };
                            AddMessageToMemory(marker);
                        }
                        catch (Exception persistError)
                        {
                            Log("Could not persist chat startup failure: " + persistError);
                            var failedIndex = _state.Jobs.FindIndex(j => j.Id == id);
                            if (failedIndex >= 0)
                                _state.Jobs[failedIndex] = _state.Jobs[failedIndex] with
                                { State = "failed", Error = "Persistenzfehler beim Chat-Start; SQLite-Zustand muss geprüft werden." };
                            _state.UpdatedAt = DateTimeOffset.UtcNow;
                            _persistenceFaulted = true;
                            _startupNotice = "WARNUNG: SQLite-Persistenz gestört. Schreibende Aktionen sind gesperrt.";
                        }
                    }
                    ChatCancels.Remove(id);
                    token.Dispose();
                    ModelGate.Release();
                    throw;
                }
            case "chat-cancel":
                if (string.IsNullOrWhiteSpace(request.Id) || !ChatCancels.TryGetValue(request.Id, out var pending))
                    return new(false, "Keine aktive Chat-Generierung mit dieser ID.");
                pending.Cancel();
                SetChatState(request.Id, "canceling", _state.Jobs.FirstOrDefault(j => j.Id == request.Id)?.Result,
                    "Abbruch angefordert; Modellressource wird freigegeben.");
                return new(true, "Abbruch läuft; die Modellressource wird noch freigegeben.", Clone());
            case "start-demo":
                if (ActiveJobCount(_state) >= MaxActiveJobs) return new(false, $"Auftragslimit erreicht ({MaxActiveJobs} gleichzeitig aktive Aufträge).");
                var started = Clone();
                started.Jobs.Add(new Job(Guid.NewGuid().ToString("N"), "SPIKE-Testauftrag: 10 Schritte",
                    "queued", 0, 10, DateTimeOffset.UtcNow));
                Save(started);
                return new(true, "Testauftrag gestartet.", Clone());
            case "cancel":
                if (string.IsNullOrWhiteSpace(request.Id)) return new(false, "Auftrags-ID fehlt.");
                var index = _state.Jobs.FindIndex(x => x.Id == request.Id);
                if (index < 0) return new(false, "Auftrag nicht gefunden.");
                if (_state.Jobs[index].State is "done" or "canceled" or "failed") return new(false, "Auftrag bereits beendet.");
                if (_state.Jobs[index].Kind == "chat")
                {
                    if (!ChatCancels.TryGetValue(request.Id, out var activeChat))
                        return new(false, "Chat-Generierung ist nicht mehr aktiv.");
                    activeChat.Cancel();
                    SetChatState(request.Id, "canceling", _state.Jobs[index].Result,
                        "Abbruch angefordert; Modellressource wird freigegeben.");
                    return new(true, "Abbruch läuft; die Modellressource wird noch freigegeben.", Clone());
                }
                var cancelled = Clone();
                cancelled.Jobs[index] = cancelled.Jobs[index] with { State = "canceled" };
                Save(cancelled);
                return new(true, "Auftrag abgebrochen.", Clone());
            case "stop":
                if (request.AuthToken is null || request.AuthToken.Length != _controlToken.Length ||
                    !System.Security.Cryptography.CryptographicOperations.FixedTimeEquals(
                        Encoding.ASCII.GetBytes(request.AuthToken), Encoding.ASCII.GetBytes(_controlToken)))
                    return new(false, "Host-Stopp nicht autorisiert.");
                _acceptingWork = false;
                return new(true, "Host wird kontrolliert beendet.");
            default: return new(false, "Unbekannter Befehl.");
        }
    }

    private static async Task WorkLoop(CancellationToken stopping)
    {
        while (!stopping.IsCancellationRequested)
        {
            try
            {
                await Task.Delay(TimeSpan.FromSeconds(1), stopping);
                lock (StateGate)
                {
                    var index = _state.Jobs.FindIndex(x => x.Kind == "demo" && (x.State is "queued" or "running"));
                    if (index >= 0)
                    {
                        var next = Clone();
                        var job = next.Jobs[index];
                        var completed = Math.Min(job.Total, job.Completed + 1);
                        next.Jobs[index] = job with
                        {
                            State = completed == job.Total ? "done" : "running",
                            Completed = completed
                        };
                        Save(next);
                    }
                }
            }
            catch (OperationCanceledException) when (stopping.IsCancellationRequested) { break; }
            catch (Exception ex)
            {
                Log("Background job paused after error: " + ex);
                try { await Task.Delay(TimeSpan.FromSeconds(5), stopping); }
                catch (OperationCanceledException) { break; }
            }
        }
    }

    private static Snapshot Clone() => new()
    {
        SchemaVersion = _state.SchemaVersion,
        Decisions = [.. _state.Decisions],
        Jobs = [.. _state.Jobs],
        ChatMessages = [.. _state.ChatMessages],
        UpdatedAt = _state.UpdatedAt
    };

    private static Snapshot ReadSnapshot(string path)
    {
        var result = JsonSerializer.Deserialize<Snapshot>(File.ReadAllText(path), Protocol.Json)
            ?? throw new InvalidDataException("Snapshot is empty.");
        if (result.SchemaVersion is not (1 or 2)) throw new InvalidDataException("Unknown state schema " + result.SchemaVersion);
        if (result.Decisions is null || result.Jobs is null) throw new InvalidDataException("Missing state collections.");
        // The legacy JSON contract is upgraded in memory before entering SQLite v2.
        result.SchemaVersion = 2;
        return result;
    }

    private static void LoadFromStorage()
    {
        try
        {
            _store = new SnapshotStore(DataDirectory);
            var current = _store.Read();
            if (current is not null)
            {
                _state = current;
                // Chat requests cannot be resumed after a process crash. Complete
                // interrupted turns so the next model context never contains a dangling user turn.
                var interruptedChats = _state.Jobs
                    .Where(j => j.Kind == "chat" && (j.State is "queued" or "running" or "canceling"))
                    .ToArray();
                foreach (var interrupted in interruptedChats)
                {
                    var wasCanceling = interrupted.State == "canceling";
                    var terminalState = wasCanceling ? "canceled" : "failed";
                    var error = wasCanceling
                        ? "Host-Neustart während des Abbruchs."
                        : "Host-Neustart während der Generierung.";
                    try
                    {
                        FinishChatLocked(interrupted.Id, interrupted.Model ?? "lokal", terminalState,
                            interrupted.Result, error,
                            wasCanceling ? "[Antwort durch Host-Neustart abgebrochen.]" :
                                "[Generierung durch Host-Neustart unterbrochen.]");
                    }
                    catch (Exception recoveryError)
                    {
                        Log("Could not persist interrupted chat recovery: " + recoveryError);
                        try
                        {
                            var now = DateTimeOffset.UtcNow;
                            _store.UpdateJobProgress(interrupted.Id, terminalState, 0, 1,
                                interrupted.Result, error, now);
                            var index = _state.Jobs.FindIndex(j => j.Id == interrupted.Id);
                            if (index >= 0) _state.Jobs[index] = _state.Jobs[index] with
                            { State = terminalState, Error = error };
                            _state.UpdatedAt = now;
                        }
                        catch (Exception fallbackError)
                        {
                            Log("Interrupted chat fallback failed: " + fallbackError);
                            throw new InvalidOperationException(
                                "GENESIS cannot safely recover an interrupted chat because SQLite is not writable.", fallbackError);
                        }
                    }
                }
                _startupNotice = "GENESIS SQLite-Zustand geladen.";
                return;
            }
        }
        catch (Exception ex) when (ex is InvalidDataException or FormatException or SqliteException)
        {
            _store?.Dispose();
            _store = null;
            var db = Path.Combine(DataDirectory, "genesis.db");
            if (File.Exists(db))
            {
                var suffix = DateTimeOffset.UtcNow.ToString("yyyyMMddHHmmssfff");
                try
                {
                    // SnapshotStore has disposed its connection before this catch. Preserve
                    // the database and any WAL/SHM sidecars without modifying the originals.
                    foreach (var candidate in new[] { db, db + "-wal", db + "-shm" })
                    {
                        if (File.Exists(candidate))
                            File.Copy(candidate, db + ".corrupt-" + suffix +
                                candidate.Substring(db.Length), overwrite: false);
                    }
                    Log("Database validation failed; preserved corrupted copy at " + db + ".corrupt-" + suffix);
                }
                catch (IOException quarantineError)
                {
                    Log("Database quarantine copy failed; original left untouched: " + quarantineError);
                }
                catch (UnauthorizedAccessException quarantineError)
                {
                    Log("Database quarantine copy denied; original left untouched: " + quarantineError);
                }
            }
            throw;
        }
        // Import is retried if the process stops before SQLite marks itself initialized.
        // Original legacy JSON and backups are NEVER deleted or overwritten by migration.
        Load();
        _store.Commit(_state);
        _state = _store.Read() ?? _state;
        _startupNotice = File.Exists(StatePath)
            ? "JSON-Zustand in SQLite migriert; Originaldateien erhalten."
            : "Neue SQLite-Datenbank initialisiert.";
    }

    private static void Load()
    {
        if (!File.Exists(StatePath))
        {
            if (File.Exists(BackupPath))
            {
                // Refuse silent loss of a previously saved snapshot.
                _state = ReadSnapshot(BackupPath);
                File.Copy(BackupPath, StatePath, overwrite: false);
                _startupNotice = "Zustand aus Backup wiederhergestellt (Hauptdatei fehlte).";
            }
            return;
        }
        try { _state = ReadSnapshot(StatePath); }
        catch (Exception ex) when (ex is IOException or JsonException or InvalidDataException)
        {
            Log("Primary state invalid: " + ex);
            if (!File.Exists(BackupPath)) throw new InvalidDataException("No backup; original preserved.", ex);
            var backup = ReadSnapshot(BackupPath); // verify backup before touching primary
            var quarantined = StatePath + ".corrupt-" + DateTimeOffset.UtcNow.ToString("yyyyMMddHHmmssfff");
            File.Move(StatePath, quarantined);
            File.Copy(BackupPath, StatePath);
            _state = backup;
            _startupNotice = "Beschädigter Zustand gesichert; Backup wiederhergestellt.";
            Log(_startupNotice + " " + quarantined);
        }
    }

    // SQLite transaction is authoritative. Publish in-memory state only after COMMIT.
    private static void Save(Snapshot candidate)
    {
        candidate.UpdatedAt = DateTimeOffset.UtcNow;
        (_store ?? throw new InvalidOperationException("SQLite is not initialized."))
            .Commit(candidate);
        TrimJobs(candidate);
        _state = candidate;
    }

    private static void Log(string message)
    {
        try
        {
            lock (LogGate)
            {
                if (File.Exists(LogPath) && new FileInfo(LogPath).Length >= MaxLogBytes)
                    File.Move(LogPath, LogPath + ".old", overwrite: true);
                File.AppendAllText(LogPath, $"{DateTimeOffset.UtcNow:u} {message}{Environment.NewLine}");
            }
        }
        catch { /* Logging must never terminate the host. */ }
    }
}

using System.IO;

namespace Genesis.Host;

public static class PipeLineCodec
{
    // A char-by-char reader bounds memory even if a client never sends a newline.
    public static async Task<string?> ReadBoundedAsync(StreamReader reader, int limit, CancellationToken ct)
    {
        if (limit < 1) throw new ArgumentOutOfRangeException(nameof(limit));
        var line = new System.Text.StringBuilder();
        var chars = new char[1];
        while (true)
        {
            var read = await reader.ReadAsync(chars.AsMemory(), ct);
            if (read == 0) return line.Length == 0 ? null : throw new InvalidDataException("Incomplete request.");
            if (chars[0] == '\n') return line.ToString().TrimEnd('\r');
            if (line.Length >= limit) throw new InvalidDataException("Request exceeds max length.");
            line.Append(chars[0]);
        }
    }
}

    private static async Task<Response> SendAsync(Request request)
    {
        using var pipe = new NamedPipeClientStream(".", Protocol.ActivePipeName, PipeDirection.InOut,
            PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
        await pipe.ConnectAsync(1500);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        using var reader = new StreamReader(pipe, Encoding.UTF8, leaveOpen: true);
        // Every IPC command, including status and model discovery, requires the current host capability.
        await writer.WriteLineAsync(JsonSerializer.Serialize(
            request with { AuthToken = request.AuthToken ?? ReadControlToken() }, Protocol.WireJson));
        using var timeout = new CancellationTokenSource(request.Command == "chat" ? TimeSpan.FromMinutes(3) : TimeSpan.FromSeconds(10));
        var reply = await ReadBoundedReplyAsync(reader, 4 * 1024 * 1024, timeout.Token);
        return reply is null ? new(false, "Keine Antwort vom Host")
            : JsonSerializer.Deserialize<Response>(reply, Protocol.WireJson)
              ?? new(false, "Ungültige Antwort");
    }

    private static async Task<string?> ReadBoundedReplyAsync(StreamReader reader, int limit, CancellationToken cancellation)
    {
        var result = new StringBuilder();
        var buffer = new char[1];
        while (true)
        {
            var count = await reader.ReadAsync(buffer.AsMemory(), cancellation);
            if (count == 0) return result.Length == 0 ? null : throw new InvalidDataException("Unvollständige Antwort.");
            if (buffer[0] == '\n') return result.ToString().TrimEnd('\r');
            if (result.Length >= limit) throw new InvalidDataException("Host-Antwort überschreitet das Größenlimit.");
            result.Append(buffer[0]);
        }
    }

    private static string? ReadControlToken()
    {
        var dataDirectory = (Protocol.TestOverridesEnabled
            ? Environment.GetEnvironmentVariable("GENESIS_DATA_DIR") : null)
            ?? Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "JoshuaSpike5Genesis");
        var path = Path.Combine(dataDirectory, "control.token");
        return File.Exists(path) ? File.ReadAllText(path).Trim() : null;
    }

    private static async Task EnsureHostAsync()
    {
        try { if ((await SendAsync(new Request("status"))).Success) return; } catch { }
        var path = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "..", "Host", "Genesis.Host.exe"));
        if (!File.Exists(path)) throw new FileNotFoundException("Host-Datei fehlt: " + path);
        using var process = Process.Start(new ProcessStartInfo(path)
        {
            UseShellExecute = false, CreateNoWindow = true
        });
        for (var i = 0; i < 15; i++)
        {
            await Task.Delay(200);
            try { if ((await SendAsync(new Request("status"))).Success) return; } catch { }
            if (process?.HasExited == true) return;
        }
    }

    private async Task RefreshAsync()
    {
        if (_refreshing) return;
        _refreshing = true;
        var generation = _stateGeneration;
        try
        {
            var response = await SendAsync(new Request("status"));
            // A user action may have produced a newer snapshot while this poll was in flight.
            if (generation != _stateGeneration) return;
            ConnectionText.Text = response.Success ? "● " + response.Message : response.Message;
            if (!response.Success) ClearStateViews();
            if (response.Data is { } data)
            {
                var runningChat = data.Jobs.LastOrDefault(j =>
                    j.Kind == "chat" && j.State is "queued" or "running" or "canceling");
                _activeChatJobId = runningChat?.Id;
                SendChatButton.IsEnabled = runningChat is null && !_sendInFlight;
                ChatCancelButton.IsEnabled = runningChat is not null && runningChat.State != "canceling";
                if (data.UpdatedAt != _lastStateRevision)
                {
                    _lastStateRevision = data.UpdatedAt;
                    DecisionsList.ItemsSource = data.Decisions.OrderByDescending(x => x.CreatedAt)
                        .Select(x => new DecisionRow(x.CreatedAt.LocalDateTime.ToString("g"), x.Text)).ToArray();
                    var selected = (JobsList.SelectedItem as JobRow)?.Id;
                    JobsList.ItemsSource = data.Jobs.OrderByDescending(x => x.CreatedAt)
                        .Select(x => new JobRow(x.Id, x.Title, x.State, x.Completed, x.Total)).ToArray();
                    JobsList.SelectedItem = JobsList.Items.Cast<JobRow>().FirstOrDefault(x => x.Id == selected);
                    RenderTranscript(data);
                }
            }
        }
        catch
        {
            if (generation != _stateGeneration) return;
            ConnectionText.Text = "○ Host nicht erreichbar – Log prüfen";
            ClearStateViews();
        }
        finally { _refreshing = false; }
    }

    // Parse a deliberately small, safe Markdown subset as WPF text runs (never HTML or executable XAML).
    private static void AppendMarkdown(Section section, string value)
    {
        var lines = value.Replace("\r\n", "\n").Replace('\r', '\n').Split('\n');
        foreach (var original in lines)
        {
            var line = original;
            var paragraph = new Paragraph { Margin = new Thickness(0, 0, 0, 4), LineHeight = 22 };
            if (line.StartsWith("### ", StringComparison.Ordinal) ||
                line.StartsWith("## ", StringComparison.Ordinal) ||
                line.StartsWith("# ", StringComparison.Ordinal))
            {
                var depth = line.TakeWhile(ch => ch == '#').Count();
                line = line[(depth + 1)..];
                paragraph.FontWeight = FontWeights.SemiBold;
                paragraph.FontSize = depth == 1 ? 19 : depth == 2 ? 17 : 15;
                paragraph.Margin = new Thickness(0, 9, 0, 6);
            }
            else if (line.StartsWith("- ", StringComparison.Ordinal) ||
                     line.StartsWith("* ", StringComparison.Ordinal))
                line = "•  " + line[2..];
            foreach (var part in Regex.Split(line, @"(\*\*[^*\r\n]+\*\*|\x60[^\x60\r\n]+\x60|\*[^*\r\n]+\*)"))
            {
                if (part.Length == 0) continue;
                if (part.Length >= 4 && part.StartsWith("**", StringComparison.Ordinal) && part.EndsWith("**", StringComparison.Ordinal))
                    paragraph.Inlines.Add(new Bold(new Run(part[2..^2])));
                else if (part.Length >= 2 && part[0] == (char)96 && part[^1] == (char)96)
                    paragraph.Inlines.Add(new Run(part[1..^1]) { FontFamily = new FontFamily("Consolas") });
                else if (part.Length >= 2 && part[0] == '*' && part[^1] == '*')
                    paragraph.Inlines.Add(new Italic(new Run(part[1..^1])));
                else paragraph.Inlines.Add(new Run(part));
            }
            section.Blocks.Add(paragraph);
        }
    }

    private static ScrollViewer? FindChatScrollViewer(DependencyObject root)

            if (child is ScrollViewer viewer) return viewer;

    private void RenderTranscript(Snapshot state)
    {
        var signature = new StringBuilder();
        foreach (var message in state.ChatMessages)
            signature.Append(message.Id).Append(':').Append(message.Text).Append('\n');
        var lastChatJob = state.Jobs.Where(j => j.Kind == "chat")
            .OrderBy(j => j.CreatedAt).LastOrDefault();
        if (lastChatJob is not null)
            signature.Append(lastChatJob.Id).Append(lastChatJob.State).Append(lastChatJob.Result).Append(lastChatJob.Error);
        var currentSignature = signature.ToString();
        if (currentSignature == _lastRenderedTranscript) return;

        var scroll = FindChatScrollViewer(ChatOutput);
        var oldOffset = scroll?.VerticalOffset ?? 0;
        var atBottom = scroll is null || scroll.ScrollableHeight - scroll.VerticalOffset < 28;
        var document = new FlowDocument
        {
            PagePadding = new Thickness(0),
            FontFamily = new FontFamily("Segoe UI"),
            FontSize = 14,
            Foreground = ChatOutput.Foreground
        };
        if (state.ChatMessages.Count == 0)
            document.Blocks.Add(new Paragraph(new Run("Noch keine Nachrichten.")));
        foreach (var message in state.ChatMessages)
        {
            var isUser = message.Role == "user";
            var section = new Section
            {
                Background = (Brush)FindResource(isUser ? "PanelBg" : "InputBg"),
                Padding = new Thickness(12, 10, 12, 8),
                Margin = new Thickness(isUser ? 35 : 0, 3, isUser ? 0 : 35, 10)
            };
            section.Blocks.Add(new Paragraph(new Run($"{(isUser ? "DU" : "JOSHUA")}  ·  {message.CreatedAt.LocalDateTime:HH:mm}  ·  {message.Model ?? "lokal"}"))
            {
                FontWeight = FontWeights.SemiBold,
                Foreground = (Brush)FindResource("Accent"),
                FontSize = 12,
                Margin = new Thickness(0, 0, 0, 7)
            });
            AppendMarkdown(section, message.Text);
            document.Blocks.Add(section);
        }
        if (lastChatJob?.State is "queued" or "running" or "canceling" or "failed")
        {
            var status = lastChatJob.State switch
            {
                "queued" => "JOSHUA · in Warteschlange",
                "running" => "JOSHUA · schreibt ...",
                "canceling" => "JOSHUA · Abbruch läuft ...",
                _ => "JOSHUA · Fehler: " + (lastChatJob.Error ?? "Generierung fehlgeschlagen")
            };
            var pending = new Section { Margin = new Thickness(0, 3, 35, 10), Padding = new Thickness(12),
                Background = (Brush)FindResource("PanelBg") };
            pending.Blocks.Add(new Paragraph(new Run(status)) { FontWeight = FontWeights.SemiBold });
            if (!string.IsNullOrEmpty(lastChatJob.Result)) AppendMarkdown(pending, lastChatJob.Result);
            document.Blocks.Add(pending);
        }
        ChatOutput.Document = document;
        _lastRenderedTranscript = currentSignature;
        ChatOutput.UpdateLayout();
        if (atBottom) ChatOutput.ScrollToEnd();
        else FindChatScrollViewer(ChatOutput)?.ScrollToVerticalOffset(oldOffset);
    }

    private async Task SendChatAsync()
    {
        if (_sendInFlight || _activeChatJobId is not null) return;
        if (ModelPicker.SelectedItem is not string model)
        {
            MessageBox.Show("Bitte zuerst ein Ollama-Modell auswählen.", "JOSHUA");
            return;
        }
        if (string.IsNullOrWhiteSpace(ChatInput.Text)) return;
        if (ChatInput.Text.Length > 8000) { MessageBox.Show("Bitte auf maximal 8.000 Zeichen kürzen.", "Chat-Limit"); return; }
        var prompt = ChatInput.Text;
        _sendInFlight = true;
        SendChatButton.IsEnabled = false;
        try
        {
            var accepted = await SendAsync(new Request("chat", Text: prompt, Model: model));
            if (!accepted.Success)
            {
                MessageBox.Show(accepted.Message, "Ollama");
                return;
            }
            ChatInput.Clear();
            // Invalidate any older timer poll that may complete after this acceptance
            // response and otherwise overwrite the newly accepted active-job state.
            _stateGeneration++;
            _activeChatJobId = accepted.Data?.Jobs.LastOrDefault(j =>
                j.Kind == "chat" && j.State is "queued" or "running" or "canceling")?.Id;
            ChatCancelButton.IsEnabled = _activeChatJobId is not null;
            SendChatButton.IsEnabled = _activeChatJobId is null;
            await RefreshAsync();
        }
        catch (Exception ex) { MessageBox.Show("Chat fehlgeschlagen: " + ex.Message, "Ollama"); }
        finally
        {
            _sendInFlight = false;
            if (_activeChatJobId is null) SendChatButton.IsEnabled = true;
        }
    }

    private void EnsureColumn(string table, string name, string definition, SqliteTransaction tx)
    {
        var exists = false;
        using (var info = _connection.CreateCommand())
        {
            info.Transaction = tx;
            info.CommandText = "PRAGMA table_info(" + table + ");";

                    exists = true;

    public Snapshot? Read()

        {

            if (GetMeta("schema_version") != Schema.ToString(CultureInfo.InvariantCulture))

            var updated = GetMeta("updated_at")
                ?? throw new InvalidDataException("SQLite updated_at metadata missing.");
            s.UpdatedAt = Date(updated, "updated_at", "metadata");

            {
                c.CommandText = "SELECT id,text,created_at FROM decisions ORDER BY created_at,id";
                using var r = c.ExecuteReader();

                    WHERE state IN ('queued','running','canceling') OR id IN (
                        SELECT id FROM jobs WHERE state NOT IN ('queued','running','canceling')
                        ORDER BY created_at DESC,id DESC LIMIT $recentLimit)
                    ORDER BY created_at,id";

                    if (string.IsNullOrWhiteSpace(id) || total < 1 || completed < 0 || completed > total)

    public void Commit(Snapshot candidate)
    {
        lock (_dbLock)
        {
            if (candidate.SchemaVersion != Schema) throw new InvalidDataException("Unsupported Snapshot contract version.");
            using var tx = _connection.BeginTransaction();
            // No full-table delete: preserve historical rows and use targeted UPSERTs.
            foreach (var d in candidate.Decisions)
            {
                using var c = _connection.CreateCommand();
                c.Transaction = tx;
                c.CommandText = @"INSERT INTO decisions(id,text,created_at) VALUES ($id,$text,$date)
                    ON CONFLICT(id) DO UPDATE SET text=excluded.text";
                c.Parameters.AddWithValue("$id", d.Id);
                c.Parameters.AddWithValue("$text", d.Text);
                c.Parameters.AddWithValue("$date", DateText(d.CreatedAt));
                c.ExecuteNonQuery();
            }
            foreach (var j in candidate.Jobs) UpsertJob(j, tx);
            SetMeta("schema_version", Schema.ToString(CultureInfo.InvariantCulture), tx);
            SetMeta("updated_at", DateText(candidate.UpdatedAt), tx);
            SetMeta("initialized", "1", tx);
            tx.Commit();
        }
    }

    public async IAsyncEnumerable<string> ChatStreamAsync(

        {
            Content = JsonContent.Create(new
            {
                model,
                stream = true,
                messages = messages.Select(m => new { role = m.Role, content = m.Content }).ToArray(),
                think = false,
                options = new { num_predict = 2048, num_ctx = 8192 }
            })

            if (root.TryGetProperty("error", out var error))
                throw new InvalidDataException("Ollama: " + (error.GetString() ?? "Fehler"));

                content.ValueKind == JsonValueKind.String)

                total += chunk.Length;
                if (total > 250000)
                    throw new InvalidDataException("Ollama output exceeded 250000 characters.");
                if (chunk.Length > 0) yield return chunk;
            }
            if (root.TryGetProperty("done", out var complete) && complete.ValueKind == JsonValueKind.True)
            {
                if (root.TryGetProperty("done_reason", out var reason) &&
                    reason.ValueKind == JsonValueKind.String &&
                    reason.GetString() is "length")
                    throw new InvalidDataException("Ollama hat die Antwort wegen des Generierungslimits gekürzt.");
                done = true;
            }
        }

    private static async Task CheckStatusAsync(HttpResponseMessage response, CancellationToken cancellation)
    {
        if (response.IsSuccessStatusCode) return;
        string detail;
        try
        {
            using var json = await ReadBoundedJsonAsync(response.Content, cancellation);
            detail = json.RootElement.TryGetProperty("error", out var error) &&
                error.ValueKind == JsonValueKind.String ? error.GetString() ?? "Fehler" : "HTTP-Fehler";
        }
        catch (Exception ex) when (ex is JsonException or InvalidDataException)
        {
            detail = "HTTP-Fehler";
        }
        throw new HttpRequestException($"Ollama HTTP {(int)response.StatusCode}: {detail}",
            null, response.StatusCode);
    }

    private static async Task<string?> ReadBoundedLineAsync(StreamReader reader, int maxChars, CancellationToken ct)
    {
        var result = new StringBuilder();
        var one = new char[1];
        while (true)
        {
            var n = await reader.ReadAsync(one.AsMemory(), ct);
            if (n == 0) return result.Length == 0 ? null
                : throw new InvalidDataException("Incomplete Ollama stream message.");
            if (one[0] == '\n') return result.ToString().TrimEnd('\r');
            if (result.Length >= maxChars) throw new InvalidDataException("Ollama stream message too large.");
            result.Append(one[0]);
        }
    }

    private static async Task<JsonDocument> ReadBoundedJsonAsync(HttpContent content, CancellationToken cancellation)
    {
        if (content.Headers.ContentLength > MaxResponseBytes)
            throw new InvalidDataException("Ollama-Antwort überschreitet das Größenlimit.");

        await using var stream = await content.ReadAsStreamAsync(cancellation);
        using var buffer = new MemoryStream();
        var chunk = new byte[8192];
        while (true)
        {
            var count = await stream.ReadAsync(chunk.AsMemory(), cancellation);
            if (count == 0) break;
            if (buffer.Length + count > MaxResponseBytes)
                throw new InvalidDataException("Ollama-Antwort überschreitet das Größenlimit.");
            buffer.Write(chunk, 0, count);
        }
        buffer.Position = 0;
        return await JsonDocument.ParseAsync(buffer, cancellationToken: cancellation);
    }

    }

async Task<Response> RequestAsync(string cmd, string? text = null, string? id = null, string? authToken = null, string? model = null)

                PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
            await client.ConnectAsync(500);
            using var writer = new StreamWriter(client, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
            using var reader = new StreamReader(client, Encoding.UTF8, leaveOpen: true);
            // Preserve the explicit unauthenticated-stop negative test; authenticate ordinary test traffic.
            var effectiveToken = authToken ?? (cmd == "stop" ? null :
                File.Exists(Path.Combine(dir, "control.token"))
                    ? File.ReadAllText(Path.Combine(dir, "control.token")).Trim() : null);
            await writer.WriteLineAsync(JsonSerializer.Serialize(new Request(cmd, text, id, effectiveToken, model), Protocol.WireJson));
            using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(10));
            var line = await reader.ReadLineAsync(timeout.Token);
            return JsonSerializer.Deserialize<Response>(line ?? throw new Exception("Null response"), Protocol.WireJson)
                ?? throw new Exception("No response object");

sealed class FakeOllama : IAsyncDisposable
{
    private readonly TcpListener _listener = new(IPAddress.Loopback, 0);
    private readonly CancellationTokenSource _stop = new();
    private readonly Task _acceptTask;
    public string Endpoint { get; }
    public volatile bool DelayChat;
    public volatile int ChatDelayMs = 5000;
    public volatile bool FailMidStream;
    private string? _lastChatBody;
    public string? LastChatBody => Volatile.Read(ref _lastChatBody);

    public FakeOllama()
    {
        _listener.Start();
        Endpoint = $"http://127.0.0.1:{((IPEndPoint)_listener.LocalEndpoint).Port}/";
        _acceptTask = Task.Run(AcceptAsync);
    }

    private async Task AcceptAsync()
    {
        while (!_stop.IsCancellationRequested)
        {
            try
            {
                var client = await _listener.AcceptTcpClientAsync(_stop.Token);
                _ = Task.Run(() => ReplyAsync(client));
            }
            catch (OperationCanceledException) { break; }
            catch (ObjectDisposedException) { break; }
        }
    }

    private async Task ReplyAsync(TcpClient client)
    {
        using (client)
        {
            try
            {
                var stream = client.GetStream();
                var header = new List<byte>();
                var byteBuffer = new byte[1];
                while (header.Count < 16384)
                {
                    if (await stream.ReadAsync(byteBuffer, _stop.Token) == 0) return;
                    header.Add(byteBuffer[0]);
                    var n = header.Count;
                    if (n >= 4 && header[n-4] == 13 && header[n-3] == 10 &&
                        header[n-2] == 13 && header[n-1] == 10) break;
                }
                var request = Encoding.ASCII.GetString(header.ToArray());
                // HttpClient may send JSON using either Content-Length or chunked framing.
                // The request body must be drained before closing, otherwise Windows
                // may reset the TCP connection while the client is reading the response.
                var headers = request.Split("\r\n", StringSplitOptions.RemoveEmptyEntries);
                var lengthLine = headers.FirstOrDefault(x => x.StartsWith("Content-Length:", StringComparison.OrdinalIgnoreCase));
                var isChunked = headers.Any(x => x.StartsWith("Transfer-Encoding:", StringComparison.OrdinalIgnoreCase)
                    && x.Contains("chunked", StringComparison.OrdinalIgnoreCase));
                using var requestBody = new MemoryStream();
                if (lengthLine is not null && int.TryParse(lengthLine[(lengthLine.IndexOf(':') + 1)..].Trim(), out var bodyLength))
                {
                    var bodyBytes = await ReadExactAsync(stream, bodyLength, _stop.Token);
                    requestBody.Write(bodyBytes);
                }
                else if (isChunked)
                {
                    while (true)
                    {
                        var lengthText = await ReadAsciiLineAsync(stream, _stop.Token);
                        var size = Convert.ToInt32(lengthText.Split(';')[0], 16);
                        if (size == 0)
                        {
                            while ((await ReadAsciiLineAsync(stream, _stop.Token)).Length > 0) { }
                            break;
                        }
                        var chunk = await ReadExactAsync(stream, size, _stop.Token);
                        requestBody.Write(chunk);
                        await ReadExactAsync(stream, 2, _stop.Token); // CRLF
                    }
                }
                var requestBodyText = Encoding.UTF8.GetString(requestBody.ToArray());
                string body;
                if (request.StartsWith("GET /api/tags", StringComparison.Ordinal))
                {
                    body = "{\"models\":[{\"name\":\"fake-alpha:1b\"}]}";
                }
                else if (request.StartsWith("POST /api/chat", StringComparison.Ordinal))
                {
                    var delayThisChat = DelayChat;
                    var failThisChat = FailMidStream;
                    Volatile.Write(ref _lastChatBody, requestBodyText);
                    if (delayThisChat) await Task.Delay(ChatDelayMs, _stop.Token);
                    if (requestBodyText.Contains("SimulateEmptyAnswer", StringComparison.Ordinal))
                        body = "{\"message\":{\"role\":\"assistant\",\"content\":\"\"},\"done\":true}\n";
                    else if (requestBodyText.Contains("SimulateLongAnswer", StringComparison.Ordinal))
                        body = "{\"message\":{\"role\":\"assistant\",\"content\":\"" +
                            new string('L', 4500) + "\"},\"done\":true}\n";
                    else if (requestBodyText.Contains("SimulateTruncatedAnswer", StringComparison.Ordinal))
                        body = "{\"message\":{\"role\":\"assistant\",\"content\":\"partial\"},\"done\":true,\"done_reason\":\"length\"}\n";
                    else
                        body = failThisChat
                            ? "{\"message\":{\"role\":\"assistant\",\"content\":\"Teilantwort\"},\"done\":false}\n" +
                              "{\"error\":\"synthetic mid-stream failure\"}\n"
                            : "{\"message\":{\"role\":\"assistant\",\"content\":\"Hallo von \"},\"done\":false}\n" +
                              "{\"message\":{\"role\":\"assistant\",\"content\":\"Fake-Ollama!\"},\"done\":true}\n";
                }
                else
                {
                    body = "{}";
                }
                var bytes = Encoding.UTF8.GetBytes(body);
                var head = Encoding.ASCII.GetBytes("HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: " +
                    bytes.Length + "\r\nConnection: close\r\n\r\n");
                await stream.WriteAsync(head, _stop.Token);
                await stream.WriteAsync(bytes, _stop.Token);
                await stream.FlushAsync(_stop.Token);
            }
            catch (Exception ex) when (ex is IOException or SocketException or OperationCanceledException) { }
        }
    }

    private static async Task<byte[]> ReadExactAsync(NetworkStream stream, int count, CancellationToken cancellation)
    {
        if (count < 0 || count > 2 * 1024 * 1024) throw new InvalidDataException("Invalid mock body length.");
        var result = new byte[count];
        var offset = 0;
        while (offset < count)
        {
            var read = await stream.ReadAsync(result.AsMemory(offset, count - offset), cancellation);
            if (read == 0) throw new EndOfStreamException();
            offset += read;
        }
        return result;
    }

    private static async Task<string> ReadAsciiLineAsync(NetworkStream stream, CancellationToken cancellation)
    {
        var bytes = new List<byte>();
        var one = new byte[1];
        while (bytes.Count < 1024)
        {
            if (await stream.ReadAsync(one, cancellation) == 0) throw new EndOfStreamException();

# PROJECT JOSHUA SPIKE 5.0 GENESIS — 0.3.2 Stability Candidate

**Status:** Review-/Alpha-Kandidat. Kein Produktions-Release. Keine lokale Freigabe vor erfolgreicher Windows-CI und unabhängiger Gegenprüfung.

## Ausgangspunkt

- Basis: GENESIS 0.3.1-R1
- Basis-Commit: `b36c3555619f2b0df4edba7d703211886004dc25`
- Ziel: die in den unabhängigen R1-Gutachten bestätigten Restfehler beheben, ohne einen unnötigen Architekturumbau vorzunehmen.
- Datenbank: weiterhin SQLite-Schema v2; keine destruktive Schemaänderung und kein Downgrade.

## Änderungen gegenüber 0.3.1-R1

1. **Persistenz-Fallback und Fail-Closed**
   - Kann ein terminaler Chat-Commit nicht gespeichert werden, versucht GENESIS eine gezielte Status-Fallback-Schreiboperation.
   - Scheitert auch diese, zeigt der In-Memory-Job ausdrücklich `failed` mit Persistenzfehler, statt fälschlich `running` zu bleiben.
   - Der Host wechselt in einen schreibgeschützten Sicherheitsmodus: Nur Statusabfrage und kontrolliertes Beenden bleiben erlaubt.
   - Scheitert auch beim Start die Wiederherstellung eines unterbrochenen Chats einschließlich Fallback, bricht der Hoststart ab, statt einen inkonsistenten Zustand zu präsentieren.

2. **Job-Limit und dauerhafte Retention**
   - Das Limit zählt aktive Jobs (`queued`, `running`, `canceling`), nicht die gesamte historische Datenbank.
   - Der Laufzeitspeicher und die UI laden die 500 jüngsten abgeschlossenen Jobs plus alle aktiven Jobs.
   - Ältere abgeschlossene Jobs bleiben vollständig in SQLite erhalten; es gibt keine automatische Löschung historischer Jobdaten.
   - Der Leseweg lädt nur das begrenzte aktuelle Fenster. Ein Index unterstützt die Status-/Zeit-Suche.

3. **Shutdown-Aufnahmesperre**
   - Ein autorisierter Stop setzt unter dem State-Lock sofort die Aufnahmesperre.
   - Bereits verbundene Clients können danach keine neuen Schreib-/Arbeitsaufträge mehr starten.
   - Der Host beendet sich anschließend kontrolliert und wartet auf aktive Client- und Chat-Tasks.

4. **UI-Senden ohne Timer-Abhängigkeit**
   - Das Command Deck übernimmt die Job-ID direkt aus der Annahmeantwort.
   - Ein lokaler In-Flight-Guard verhindert Doppelsenden während der Pipe-Anfrage.
   - Ein übersprungener Timer-Refresh kann den Senden-Button nicht mehr vorzeitig freigeben.
   - Status-Polls, die vor einer neueren Chat-Annahme gestartet wurden, dürfen deren aktiven Zustand nicht überschreiben.
   - Bei vorübergehend nicht erreichbarem Host wird eine bekannte aktive Job-ID nicht voreilig verworfen.

5. **Robuste Streaming-Fortschrittsupdates**
   - Ein SQLite-Fehler bei einem nicht-terminalen Fortschrittsupdate wird protokolliert; der Ollama-Stream läuft weiter.
   - Die abschließende Transaktion bleibt maßgeblich. Scheitert diese ebenfalls, greifen Fallback und Fail-Closed.

6. **Vollständiger Beschluss- und Gesprächskontext**
   - Beschlüsse und Gesprächsrunden werden nur vollständig übernommen; keine Teilstrings mitten im Satz.
   - Neueste Beschlüsse und jüngste vollständige Gesprächspaare haben Vorrang.
   - Ausgelassene ältere oder zu große Inhalte werden ausdrücklich im Systemkontext und Statushinweis ausgewiesen.
   - Zeichenbudgets bleiben Alpha-Grenzen und sind keine echte Tokenzählung.

## Neue Regressionstests

- 505 abgeschlossene Jobs plus ein aktiver Job: Retention bewahrt genau die 500 neuesten abgeschlossenen und den aktiven Job.
- Neue Chat-Erstellung führt Retention atomar fort; aktive Jobs bleiben erhalten.
- Beschlusskontext: neueste vollständige Beschlüsse werden übertragen; ältere, nicht passende Beschlüsse werden nicht abgeschnitten und ihre Auslassung wird ausgewiesen.
- Authentifizierter Host-Stop: neue Chats werden nach gesetzter Aufnahmesperre abgewiesen.
- Doppelte terminale Persistenzfehler: UI-/Statuszustand bleibt nicht fälschlich `running`; weitere Schreibaktionen werden gesperrt; ein normaler Neustart repariert den offenen Datenbankjob.
- Test-only injizierter einmaliger Fehler eines nicht-terminalen Fortschrittsupdates während des Ollama-Streams: die Generierung muss weiterlaufen und die finale Antwort erfolgreich gespeichert werden. Das ist ein deterministischer Regressionstest des Fehlerpfads, kein echter Mehrprozess-SQLite-Lock-Benchmark.
- Bestehende Migration-, Atomic-Create-, Cancel-, Restart-, Pipe-, Quarantäne- und Fake-Ollama-Regressionstests bleiben Bestandteil des Laufs.

Die künstlichen Fehler-Injektionen (terminaler Commit, Fallback und einmaliges Fortschrittsupdate) werden ausschließlich bei aktiviertem Test-Override (`GENESIS_ALLOW_TEST_OVERRIDES=1`) und nur über explizite Testvariablen aktiviert. Sie sind in normaler Ausführung nicht aktiv.

## Abnahme- und Sicherheitsgrenzen

- Vor einem lokalen Test den gesamten Ordner `%LOCALAPPDATA%\\JoshuaSpike5Genesis` bei beendetem Host sichern, einschließlich vorhandener SQLite-WAL-/SHM-Dateien.
- Keine Kompatibilität mit älteren Hosts nach Schema-Downgrade versprechen.
- Kein echter Qwen-/Ollama-Lauf auf der RTX 4070 SUPER durch GitHub Actions.
- Keine interaktive WPF-Abnahme, keine Drei-Monitor-/4K-TV-/Misch-DPI-Abnahme durch CI.
- Kein mehrstündiger Soak-Test durch CI.
- Kein unbeaufsichtigter Dauerbetrieb, bevor die manuelle Zielrechner-Abnahme abgeschlossen ist.

## Freigabekriterien

1. Alle Projekte bauen unter Windows x64.
2. Protokoll- und Ollama-Vertragstests bestehen.
3. Vollständige Host-Integrationstests bestehen einschließlich der neuen Retention-, Fail-Closed-, Shutdown-, Kontext- und SQLite-Lock-Szenarien.
4. Das veröffentlichte CI-Artefakt stammt exakt vom geprüften Commit.

6. Erst danach kontrollierter lokaler Test mit Backup und Beobachtung.

# GENESIS 0.3.2-R1 — Foundation Hardening: Windows-Alpha-Abnahme

**Entwurfsstatus.** Kein Produktionsrelease, kein automatischer Freigabebeschluss. Build und Regressionstests müssen zum exakten Commit erfolgreich sein. Vor der lokalen Alpha: kompletten Ordner `%LOCALAPPDATA%\JoshuaSpike5Genesis` mit gestopptem Host sichern (einschließlich WAL-/SHM-Sidecars).

## Ziel dieser Korrekturrunde

Die unabhängigen 0.3.2-Gutachten meldeten vier gewichtige Restpunkte: Host-Stopp bei unterbrochener Pipe, leere Ollama-Antwort, Abbruch der Historienauswahl bei einem zu langen Austausch und große Statusantworten. Daneben wurden doppelte Systemnachrichten, fehlende Ollama-`done_reason`-Auswertung, fehlender Kontextparameter, irreführende UI-Version und eine nicht belegte Lost-Update-Behauptung genannt.

**Quellcode-Gegenprüfung:** `Execute()` wird bereits in `HandleClient` unter `lock(StateGate)` ausgeführt. Für diese konkrete Befehlsverarbeitung besteht daher nicht der im Gutachten behauptete ungesperrte Schreibpfad. Der neue Regressionstest sendet zwei `remember`-Befehle gleichzeitig.

## Implementierte Korrekturen

- **Stopp:** `StopSignal.Cancel()` wird im `finally` einer bestätigten Stop-Antwort ausgelöst, auch wenn das Zurückschreiben zum Client scheitert. Der Chat-Shutdown erhält eine Wartegrenze von 15 Sekunden; danach beendet sich der Hostprozess statt unbegrenzt zu warten.
- **Wire-Antwort:** Jeder Snapshot für die UI wird als eigenständige reduzierte Kopie übertragen. Historische Job-`Prompt`-/`Result`-Texte werden nicht mehr pro Sekunde mitgeschickt. Der letzte abgeschlossene Chat darf seine Antwort bis 12.000 Zeichen zeigen, ebenso laufende Chat-Teiltexte. Von den Chatnachrichten werden die letzten 40 (jeweils höchstens 12.000 Zeichen) dargestellt. Die vollständigen Daten bleiben in SQLite.
- **Modellantwort:** Eine komplett leere Ollama-Antwort wird als `failed` statt `done` gespeichert. Wenn Ollama `done_reason=length` meldet, wird die Antwort als abgebrochen/gekürzt und nicht als vollständiger Erfolg behandelt.
- **Kontext:** Alte vollständige Gesprächsrunden werden nicht sämtlich verworfen, wenn die neueste Antwort das Geschichtsbudget überschreitet. Fehlermarker und Cancel-Marker werden nicht als scheinbare Assistenzantworten zurück in das Modell gegeben. Systemkontext und Budgethinweis werden zu genau einem System-Turn zusammengefügt. Gespeicherte Beschlüsse werden ausdrücklich als Nutzerdaten gekennzeichnet, nicht als übergeordnete Systeminstruktionen.
- **Ollama:** `think=false`, `num_ctx=8192` und `num_predict=2048` werden explizit übergeben. Die nicht-streamende Methode verarbeitet strukturierte Ollama-Fehler und leere Antworten.
- **Oberfläche:** Fenstertitel und Kopfzeile tragen `0.3.2-R1`.

## Neue automatisierte Regressionstests

- Zwei gleichzeitige `remember`-Aufträge erhalten beide Änderungen.
- Fake-Ollama meldet `done=true` ohne sichtbaren Text: Job muss `failed` werden.
- Eine 4.500-Zeichen-Antwort darf nicht alle früheren abgeschlossenen Gespräche aus dem Modellkontext verdrängen.
- Fehlermarker werden nicht wieder in Modellnachrichten eingespeist.
- Nur eine Systemnachricht und explizite `think`/`num_ctx`-Optionen.
- `done_reason=length` muss als Fehler statt als normale Antwort gespeichert werden.
- Host-Stopp trotz Client-Disconnect vor der Bestätigung.
- Historische Jobs mit sehr großen Prompts/Resultaten: der UI-Status bleibt unter dem 4-MiB-Limit.

Bestehende SQLite-Migrations-, Atomizitäts-, Recovery-, Abbruch-, Quarantäne- und Host-Integrationstests bleiben unverzichtbar.

## Grenzen des Nachweises

- Fake-Ollama ist **kein echter Qwen-/Ollama-GPU-Test**. Die CLI- und WPF-UI muss auf dem Windows-11-PC interaktiv geprüft werden; 12-GB-VRAM, lange Inferenz, Modellladezeiten und echte HTTP-Chunk-Verzögerungen sind nicht durch diesen Build nachgewiesen.
