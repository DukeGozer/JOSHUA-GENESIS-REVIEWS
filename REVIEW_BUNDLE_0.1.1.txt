using System.IO.Pipes;
using System.Text;
using System.Text.Json;
using Genesis.Contracts;

namespace Genesis.Host;

internal static class Program
{
    private static readonly object StateGate = new();
    private static readonly object LogGate = new();
    private static readonly CancellationTokenSource StopSignal = new();
    private static readonly string DataDirectory = Environment.GetEnvironmentVariable("GENESIS_DATA_DIR")
        ?? Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "JoshuaSpike5Genesis");
    private static readonly string StatePath = Path.Combine(DataDirectory, "state.json");
    private static readonly string BackupPath = StatePath + ".bak";
    private static readonly string LogPath = Path.Combine(DataDirectory, "host.log");
    private static Snapshot _state = new();
    private static string _startupNotice = "GENESIS Host erreichbar";
    private const int MaxRequestChars = 64 * 1024;

    private static async Task<int> Main()
    {
        Directory.CreateDirectory(DataDirectory);
        try
        {
            using var singleton = new FileStream(Path.Combine(DataDirectory, "host.lock"),
                FileMode.OpenOrCreate, FileAccess.ReadWrite, FileShare.None);
            Load();
            Log("GENESIS 0.1.1 Host started.");
            var worker = Task.Run(() => WorkLoop(StopSignal.Token));
            try
            {
                while (!StopSignal.IsCancellationRequested)
                {
                    using var server = new NamedPipeServerStream(Protocol.ActivePipeName,
                        PipeDirection.InOut, NamedPipeServerStream.MaxAllowedServerInstances,
                        PipeTransmissionMode.Byte, PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
                    try
                    {
                        await server.WaitForConnectionAsync(StopSignal.Token);
                        // Ownership of each connected pipe transfers to its request task.
                        var client = server;
                        await HandleClient(client, StopSignal.Token);
                    }
                    catch (OperationCanceledException) when (StopSignal.IsCancellationRequested) { break; }
                    catch (Exception ex)
                    {
                        Log("Pipe accept/handle error: " + ex);
                        if (!StopSignal.IsCancellationRequested) await Task.Delay(200);
                    }
                }
            }
            finally
            {
                StopSignal.Cancel();
                try { await worker; } catch (OperationCanceledException) { }
                Log("GENESIS host stopped.");
            }
            return 0;
        }
        catch (IOException ex) { Log("Host lock unavailable (possibly already running): " + ex.Message); return 2; }
        catch (Exception ex) { Log("Host startup failed: " + ex); return 1; }
    }

    private static async Task HandleClient(NamedPipeServerStream pipe, CancellationToken stopping)
    {
        // Handle one request per connection; every read has a deadline and size bound.
        using var reader = new StreamReader(pipe, Encoding.UTF8, detectEncodingFromByteOrderMarks: false, leaveOpen: true);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        try
        {
            using var timeout = CancellationTokenSource.CreateLinkedTokenSource(stopping);
            timeout.CancelAfter(TimeSpan.FromSeconds(10));
            var line = await PipeLineCodec.ReadBoundedAsync(reader, MaxRequestChars, timeout.Token);
            if (line is null) return;
            Request? request;
            try { request = JsonSerializer.Deserialize<Request>(line, Protocol.WireJson); }
            catch (JsonException)
            {
                await writer.WriteLineAsync(JsonSerializer.Serialize(
                    new Response(false, "Ungültiges JSON."), Protocol.WireJson));
                return;
            }

            Response response;
            lock (StateGate)
            {
                try { response = Execute(request); }
                catch (IOException ex) { Log("Persistence I/O error: " + ex); response = new(false, "Speichern fehlgeschlagen. Keine Änderung bestätigt."); }
                catch (UnauthorizedAccessException ex) { Log("Persistence permission error: " + ex); response = new(false, "Speichern nicht erlaubt."); }
            }
            await writer.WriteLineAsync(JsonSerializer.Serialize(response, Protocol.WireJson));
            if (request?.Command?.Equals("stop", StringComparison.OrdinalIgnoreCase) == true && response.Success)
                StopSignal.Cancel();
        }
        catch (OperationCanceledException) when (!stopping.IsCancellationRequested)
        {
            await TryReplyAsync(writer, new(false, "Zeitüberschreitung beim Lesen."));
        }
        catch (InvalidDataException) { await TryReplyAsync(writer, new(false, "Anfrage zu groß.")); }
        catch (OperationCanceledException) when (stopping.IsCancellationRequested) { }
        catch (Exception ex) { Log("Client error: " + ex); await TryReplyAsync(writer, new(false, "Anfrage fehlgeschlagen.")); }
    }

    private static async Task TryReplyAsync(StreamWriter writer, Response response)
    {
        try { await writer.WriteLineAsync(JsonSerializer.Serialize(response, Protocol.WireJson)); }
        catch (IOException) { }
        catch (ObjectDisposedException) { }
    }

    private static Response Execute(Request? request)
    {
        if (string.IsNullOrWhiteSpace(request?.Command)) return new(false, "Befehl fehlt.");
        switch (request.Command.Trim().ToLowerInvariant())
        {
            case "status": return new(true, _startupNotice, Clone());
            case "remember":
                if (string.IsNullOrWhiteSpace(request.Text)) return new(false, "Bitte Beschlusstext angeben.");
                if (request.Text.Length > 2000) return new(false, "Beschlusstext zu lang (max. 2000 Zeichen).");
                var decided = Clone();
                decided.Decisions.Add(new Decision(Guid.NewGuid().ToString("N"), request.Text.Trim(), DateTimeOffset.UtcNow));
                Save(decided);
                return new(true, "Beschluss gespeichert.", Clone());
            case "start-demo":
                var started = Clone();
                started.Jobs.Add(new Job(Guid.NewGuid().ToString("N"), "SPIKE-Testauftrag: 10 Schritte",
                    "queued", 0, 10, DateTimeOffset.UtcNow));
                Save(started);
                return new(true, "Testauftrag gestartet.", Clone());
            case "cancel":
                if (string.IsNullOrWhiteSpace(request.Id)) return new(false, "Auftrags-ID fehlt.");
                var index = _state.Jobs.FindIndex(x => x.Id == request.Id);
                if (index < 0) return new(false, "Auftrag nicht gefunden.");
                if (_state.Jobs[index].State is "done" or "canceled") return new(false, "Auftrag bereits beendet.");
                var cancelled = Clone();
                cancelled.Jobs[index] = cancelled.Jobs[index] with { State = "canceled" };
                Save(cancelled);
                return new(true, "Auftrag abgebrochen.", Clone());
            case "stop": return new(true, "Host wird kontrolliert beendet.");
            default: return new(false, "Unbekannter Befehl.");
        }
    }

    private static async Task WorkLoop(CancellationToken stopping)
    {
        while (!stopping.IsCancellationRequested)
        {
            try
            {
                await Task.Delay(TimeSpan.FromSeconds(1), stopping);
                lock (StateGate)
                {
                    var index = _state.Jobs.FindIndex(x => x.State is "queued" or "running");
                    if (index >= 0)
                    {
                        var next = Clone();
                        var job = next.Jobs[index];
                        var completed = Math.Min(job.Total, job.Completed + 1);
                        next.Jobs[index] = job with
                        {
                            State = completed == job.Total ? "done" : "running",
                            Completed = completed
                        };
                        Save(next);
                    }
                }
            }
            catch (OperationCanceledException) when (stopping.IsCancellationRequested) { break; }
            catch (Exception ex)
            {
                Log("Background job paused after error: " + ex);
                try { await Task.Delay(TimeSpan.FromSeconds(5), stopping); }
                catch (OperationCanceledException) { break; }
            }
        }
    }

    private static Snapshot Clone() => new()
    {
        SchemaVersion = _state.SchemaVersion,
        Decisions = [.. _state.Decisions],
        Jobs = [.. _state.Jobs],
        UpdatedAt = _state.UpdatedAt
    };

    private static Snapshot ReadSnapshot(string path)
    {
        var result = JsonSerializer.Deserialize<Snapshot>(File.ReadAllText(path), Protocol.Json)
            ?? throw new InvalidDataException("Snapshot is empty.");
        if (result.SchemaVersion != 1) throw new InvalidDataException("Unknown state schema " + result.SchemaVersion);
        if (result.Decisions is null || result.Jobs is null) throw new InvalidDataException("Missing state collections.");
        return result;
    }

    private static void Load()
    {
        if (!File.Exists(StatePath))
        {
            if (File.Exists(BackupPath))
            {
                // Refuse silent loss of a previously saved snapshot.
                _state = ReadSnapshot(BackupPath);
                File.Copy(BackupPath, StatePath);
                _startupNotice = "Zustand aus Backup wiederhergestellt (Hauptdatei fehlte).";
            }
            return;
        }
        try { _state = ReadSnapshot(StatePath); }
        catch (Exception ex) when (ex is IOException or JsonException or InvalidDataException)
        {
            Log("Primary state invalid: " + ex);
            if (!File.Exists(BackupPath)) throw new InvalidDataException("No backup; original preserved.", ex);
            var backup = ReadSnapshot(BackupPath); // verify backup before touching primary
            var quarantined = StatePath + ".corrupt-" + DateTimeOffset.UtcNow.ToString("yyyyMMddHHmmssfff");
            File.Move(StatePath, quarantined);
            File.Copy(BackupPath, StatePath);
            _state = backup;
            _startupNotice = "Beschädigter Zustand gesichert; Backup wiederhergestellt.";
            Log(_startupNotice + " " + quarantined);
        }
    }

    // Commit to disk before making the new state visible in memory.
    private static void Save(Snapshot candidate)
    {
        candidate.UpdatedAt = DateTimeOffset.UtcNow;
        var temp = StatePath + ".tmp";
        using (var stream = new FileStream(temp, FileMode.Create, FileAccess.Write, FileShare.None))
        {
            var bytes = Encoding.UTF8.GetBytes(JsonSerializer.Serialize(candidate, Protocol.Json));
            stream.Write(bytes);
            stream.Flush(flushToDisk: true);
        }
        if (File.Exists(StatePath))
            File.Replace(temp, StatePath, BackupPath);
        else
            File.Move(temp, StatePath);
        _state = candidate;
    }

    private static void Log(string message)
    {
        try { lock (LogGate) File.AppendAllText(LogPath, $"{DateTimeOffset.UtcNow:u} {message}{Environment.NewLine}"); }
        catch { /* Logging must never terminate the host. */ }
    }
}

using System.IO;

namespace Genesis.Host;

public static class PipeLineCodec
{
    // A char-by-char reader bounds memory even if a client never sends a newline.
    public static async Task<string?> ReadBoundedAsync(StreamReader reader, int limit, CancellationToken ct)
    {
        if (limit < 1) throw new ArgumentOutOfRangeException(nameof(limit));
        var line = new System.Text.StringBuilder();
        var chars = new char[1];
        while (true)
        {
            var read = await reader.ReadAsync(chars.AsMemory(), ct);
            if (read == 0) return line.Length == 0 ? null : throw new InvalidDataException("Incomplete request.");
            if (chars[0] == '\n') return line.ToString().TrimEnd('\r');
            if (line.Length >= limit) throw new InvalidDataException("Request exceeds max length.");
            line.Append(chars[0]);
        }
    }
}

using System.Diagnostics;
using System.IO;
using System.IO.Pipes;
using System.Text;
using System.Text.Json;
using System.Windows;
using System.Windows.Threading;
using Genesis.Contracts;

namespace Genesis.UI;

public partial class MainWindow : Window
{
    private readonly DispatcherTimer _timer = new() { Interval = TimeSpan.FromSeconds(1) };
    private bool _refreshing;
    private sealed record JobRow(string Id, string Title, string State, int Completed, int Total)
    {
        public override string ToString() =>
            $"{Title}\n{State} – {Completed}/{Total}  ·  {(Id.Length > 8 ? Id[..8] : Id)}";
    }

    public MainWindow()
    {
        InitializeComponent();
        Loaded += async (_, _) => { await EnsureHostAsync(); await RefreshAsync(); _timer.Start(); };
        Closed += (_, _) => _timer.Stop(); // Closing the UI MUST NOT terminate the Host.
        _timer.Tick += async (_, _) => await RefreshAsync();
    }

    private static async Task<Response> SendAsync(Request request)
    {
        using var pipe = new NamedPipeClientStream(".", Protocol.ActivePipeName, PipeDirection.InOut,
            PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
        await pipe.ConnectAsync(1500);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        using var reader = new StreamReader(pipe, Encoding.UTF8, leaveOpen: true);
        await writer.WriteLineAsync(JsonSerializer.Serialize(request, Protocol.WireJson));
        using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(10));
        var reply = await reader.ReadLineAsync(timeout.Token);
        return reply is null ? new(false, "Keine Antwort vom Host")
            : JsonSerializer.Deserialize<Response>(reply, Protocol.WireJson)
              ?? new(false, "Ungültige Antwort");
    }

    private static async Task EnsureHostAsync()
    {
        try { if ((await SendAsync(new Request("status"))).Success) return; } catch { }
        var path = Path.GetFullPath(Path.Combine(AppContext.BaseDirectory, "..", "Host", "Genesis.Host.exe"));
        if (!File.Exists(path)) return;
        using var process = Process.Start(new ProcessStartInfo(path)
        {
            UseShellExecute = false, CreateNoWindow = true
        });
