sealed class FakeOllama : IAsyncDisposable
{
    private readonly TcpListener _listener = new(IPAddress.Loopback, 0);
    private readonly CancellationTokenSource _stop = new();
    private readonly Task _acceptTask;
    public string Endpoint { get; }
    public volatile bool DelayChat;

    public FakeOllama()
    {
        _listener.Start();
        Endpoint = $"http://127.0.0.1:{((IPEndPoint)_listener.LocalEndpoint).Port}/";
        _acceptTask = Task.Run(AcceptAsync);
    }

    private async Task AcceptAsync()
    {
        while (!_stop.IsCancellationRequested)
        {
            try
            {
                var client = await _listener.AcceptTcpClientAsync(_stop.Token);
                _ = Task.Run(() => ReplyAsync(client));
            }
            catch (OperationCanceledException) { break; }
            catch (ObjectDisposedException) { break; }
        }
    }

    private async Task ReplyAsync(TcpClient client)
    {
        using (client)
        {
            try
            {
                var stream = client.GetStream();
                var header = new List<byte>();
                var byteBuffer = new byte[1];
                while (header.Count < 16384)
                {
                    if (await stream.ReadAsync(byteBuffer, _stop.Token) == 0) return;
                    header.Add(byteBuffer[0]);
                    var n = header.Count;
                    if (n >= 4 && header[n-4] == 13 && header[n-3] == 10 &&
                        header[n-2] == 13 && header[n-1] == 10) break;
                }
                var request = Encoding.ASCII.GetString(header.ToArray());
                // HttpClient may send JSON using either Content-Length or chunked framing.
                // The request body must be drained before closing, otherwise Windows
                // may reset the TCP connection while the client is reading the response.
                var headers = request.Split("\r\n", StringSplitOptions.RemoveEmptyEntries);
                var lengthLine = headers.FirstOrDefault(x => x.StartsWith("Content-Length:", StringComparison.OrdinalIgnoreCase));
                var isChunked = headers.Any(x => x.StartsWith("Transfer-Encoding:", StringComparison.OrdinalIgnoreCase)
                    && x.Contains("chunked", StringComparison.OrdinalIgnoreCase));
                if (lengthLine is not null && int.TryParse(lengthLine[(lengthLine.IndexOf(':') + 1)..].Trim(), out var bodyLength))
                    await ReadExactAsync(stream, bodyLength, _stop.Token);
                else if (isChunked)
                {
                    while (true)
                    {
                        var lengthText = await ReadAsciiLineAsync(stream, _stop.Token);
                        var size = Convert.ToInt32(lengthText.Split(';')[0], 16);
                        if (size == 0)
                        {
                            while ((await ReadAsciiLineAsync(stream, _stop.Token)).Length > 0) { }
                            break;
                        }
                        await ReadExactAsync(stream, size, _stop.Token);
                        await ReadExactAsync(stream, 2, _stop.Token); // CRLF
                    }
                }
                string body;
                if (request.StartsWith("GET /api/tags", StringComparison.Ordinal))
                {
                    body = "{\"models\":[{\"name\":\"fake-alpha:1b\"}]}";
                }
                else if (request.StartsWith("POST /api/chat", StringComparison.Ordinal))
                {
                    if (DelayChat) await Task.Delay(5000, _stop.Token);
                    body = "{\"message\":{\"role\":\"assistant\",\"content\":\"Hallo von \"},\"done\":false}\n" +
                           "{\"message\":{\"role\":\"assistant\",\"content\":\"Fake-Ollama!\"},\"done\":true}\n";
                }
                else
                {
                    body = "{}";
                }
                var bytes = Encoding.UTF8.GetBytes(body);
                var head = Encoding.ASCII.GetBytes("HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: " +
                    bytes.Length + "\r\nConnection: close\r\n\r\n");
                await stream.WriteAsync(head, _stop.Token);
                await stream.WriteAsync(bytes, _stop.Token);
                await stream.FlushAsync(_stop.Token);
            }
            catch (Exception ex) when (ex is IOException or SocketException or OperationCanceledException) { }
        }
    }

    private static async Task ReadExactAsync(NetworkStream stream, int count, CancellationToken cancellation)
    {
        if (count < 0 || count > 2 * 1024 * 1024) throw new InvalidDataException("Invalid mock body length.");
        var buffer = new byte[8192];
        while (count > 0)
        {
            var read = await stream.ReadAsync(buffer.AsMemory(0, Math.Min(count, buffer.Length)), cancellation);
            if (read == 0) throw new EndOfStreamException();
            count -= read;
        }
    }

    private static async Task<string> ReadAsciiLineAsync(NetworkStream stream, CancellationToken cancellation)
    {
        var bytes = new List<byte>();
        var one = new byte[1];
        while (bytes.Count < 1024)
        {
            if (await stream.ReadAsync(one, cancellation) == 0) throw new EndOfStreamException();
            if (one[0] == 10) return Encoding.ASCII.GetString(bytes.ToArray()).TrimEnd('\r');
            bytes.Add(one[0]);
        }
        throw new InvalidDataException("Mock HTTP header line too large.");
    }

    public async ValueTask DisposeAsync()
    {
        _stop.Cancel();
        _listener.Stop();
        try { await _acceptTask; } catch (OperationCanceledException) { }
        _stop.Dispose();
    }
}

sealed class MockOllamaHandler : HttpMessageHandler
{
    public bool SawPost { get; private set; }
    public string? ChatReplyOverride { get; set; }
    public bool ForceMissingModel { get; set; }
    public string? ChatModel { get; private set; }
    public string? ChatPrompt { get; private set; }

    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
    {
        if (request.Method == HttpMethod.Get && request.RequestUri?.AbsolutePath == "/api/tags")
        {
            return Ok("""{"models":[{"name":"qwen3.5:9b"},{"name":"llama3.2:3b"}]}""");
        }
        if (request.Method == HttpMethod.Post && request.RequestUri?.AbsolutePath == "/api/chat")
        {
            SawPost = true;
            var raw = await request.Content!.ReadAsStringAsync(cancellationToken);
            using var doc = JsonDocument.Parse(raw);
            var streaming = doc.RootElement.GetProperty("stream").GetBoolean();
            ChatModel = doc.RootElement.GetProperty("model").GetString();
            var messages = doc.RootElement.GetProperty("messages");
            ChatPrompt = messages[0].GetProperty("content").GetString();
            if (messages[0].GetProperty("role").GetString() is not ("user" or "system")) throw new Exception("Invalid first role");
            if (ForceMissingModel) return new HttpResponseMessage(HttpStatusCode.NotFound)
                { Content = new StringContent("""{"error":"model not found"}""", Encoding.UTF8, "application/json") };
            if (streaming)
                return Ok("""{"message":{"content":"Hallo "},"done":false}""" + "\n" +
                    """{"message":{"content":"Streaming!"},"done":true}""" + "\n");
            return Ok(ChatReplyOverride ?? """{"message":{"role":"assistant","content":"Hallo! Ich bin lokal."},"done":true}""");
        }
        return new HttpResponseMessage(HttpStatusCode.NotFound);
    }

    private static HttpResponseMessage Ok(string payload) =>
        new(HttpStatusCode.OK) { Content = new StringContent(payload, Encoding.UTF8, "application/json") };
}

# GENESIS 0.3.1 — Windows Alpha-Abnahme (Entwurf)

**Kein Produktions-Release.** Die installierbare Windows-Testfassung darf erst nach erfolgreichem CI-Lauf und manueller Abnahme als Alpha bezeichnet werden.

## Was 0.3.1 enthält

- SQLite schema v2 mit transaktionaler Migration von schema v1; `chat_messages` als Append-only-Tabelle
- Chat als dauerhafter Hintergrundauftrag: kurze Named-Pipe-Antwort mit Job-ID, Polling über status
- Eine aktive lokale Ollama-Generierung; zusätzliche Chats werden bei Auslastung abgelehnt
- Cancel per `chat-cancel`, Neustart kennzeichnet abgebrochene Host-Generierung als fehlgeschlagen
- Token-Streaming von Ollama mit Zwischenergebnis im Chat-Auftrag (Status-Abfrage)
- Verlauf, letzte 12 Nachrichten und letzte 12 Beschlüsse als Kontext (mit Längenbegrenzungen)
- WPF Command Deck: Light/Dark, Enter=Senden, Shift+Enter=Zeilenumbruch, Abbrechen, Kopieren, Fehleranzeige
- Konsolenfreier Launcher direkt im Stammordner
- Host-Integrationstest gegen isolierten lokalen Fake-Ollama-Server und SQLite-Fehlerfälle

## Start der Testfassung

1. Windows 11 x64 und .NET 8 Desktop Runtime installieren. Ollama lokal starten und ein passendes Modell vorhalten (beispielsweise `qwen3.5:9b`).
2. GitHub Actions im privaten Entwicklungsrepo: nur **erfolgreichen** Lauf für den exakten PR-Commit verwenden. Artefakt `GENESIS-0.3.1-win-x64-UNREVIEWED` herunterladen und vollständig entpacken.
3. Im entpackten Stammordner `Genesis.Launcher.exe` doppelklicken. **Nicht** nur die EXE ohne die `UI/`- und `Host/`-Unterordner kopieren.
4. Command Deck: `Modelle laden`, Modell auswählen, Nachricht eingeben, Enter senden. Status/Teilantwort und Abbrechen prüfen.
5. Eine zweite Nachricht senden und kontrollieren, ob die erste im Kontext berücksichtigt wird. Host beenden, GUI erneut starten, Verlauf prüfen.
6. Während einer langen Antwort auf `Abbrechen` klicken. Statusabfrage/andere Bedienung dürfen nicht hängen bleiben.
7. Ollama bewusst beenden und Fehleranzeige/Host-Protokoll prüfen. Kein stilles Hängen.
8. Light/Dark und Fenstergröße auf den beiden PC-Monitoren sowie dem 4K-TV bei unterschiedlichen DPI-Skalierungen überprüfen; Screenshots und Auffälligkeiten notieren.

## Schutz bestehender Daten

Der neue Host verwendet standardmäßig `%LOCALAPPDATA%\JoshuaSpike5Genesis\genesis.db`. Vor einem Alpha-Test mit wertvollen älteren GENESIS-Daten diesen **gesamten Ordner bei beendetem Host** sichern, einschließlich `.db-wal` und `.db-shm` falls vorhanden. Die Migration 1→2 ist aufwärtsgerichtet: ein 0.3.0-Host ist nicht kompatibel mit schema v2. **Kein automatischer Downgrade.**

Alte `state.json`-Quelldateien werden bei Erstimport nicht gelöscht. Ungültige SQLite-Datumswerte führen zu Fehler und einer Quarantäne-Kopie statt stiller Neuinitialisierung. Eine Quarantäne ist noch keine automatische Wiederherstellung.

## Automatisch geprüfte Kriterien

- Host/UI/Launcher kompilieren und Paket wird vollständig veröffentlicht
- Protokoll, Fehlerfälle und Ollama-Streaming (Mock)
- JSON→SQLite und SQLite-schema-v1→v2 Migration
- Chat über Named Pipe mit Fake-Ollama, Auftragserstellung, Parallelstatus, Abbruch
- Gesprächsverlauf über Host-Neustart hinweg; beschädigte Daten werden erkannt

**Noch ausstehend:** realer Ollama-/Qwen-Durchlauf, visuelle Interaktion mit WPF, GPU-Auslastungs-/12-GB-VRAM-Messung und mehrstündiger Dauerbetrieb auf dem Windows-Rechner. Github Actions testet keinen echten lokalen Qwen-Prozess.
