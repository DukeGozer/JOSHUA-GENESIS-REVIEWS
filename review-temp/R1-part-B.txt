using System.Globalization;
using Genesis.Contracts;
using Microsoft.Data.Sqlite;

namespace Genesis.Storage;

public sealed class SnapshotStore : IDisposable
{
    private readonly SqliteConnection _connection;
    private readonly object _dbLock = new();
    private const int Schema = 2;
    private readonly string _path;

    public SnapshotStore(string dataDirectory)
    {
        Directory.CreateDirectory(dataDirectory);
        _path = Path.Combine(dataDirectory, "genesis.db");
        _connection = new SqliteConnection(new SqliteConnectionStringBuilder
        {
            DataSource = _path, Mode = SqliteOpenMode.ReadWriteCreate,
            Cache = SqliteCacheMode.Private, Pooling = false
        }.ToString());
        try
        {
            _connection.Open();
            Exec("PRAGMA journal_mode=WAL;");
            Exec("PRAGMA busy_timeout=5000;");
            Exec("PRAGMA synchronous=FULL;");
            Exec("CREATE TABLE IF NOT EXISTS metadata (key TEXT PRIMARY KEY, value TEXT NOT NULL);");
            Exec("CREATE TABLE IF NOT EXISTS decisions (id TEXT PRIMARY KEY, text TEXT NOT NULL, created_at TEXT NOT NULL);");
            Exec("CREATE TABLE IF NOT EXISTS jobs (id TEXT PRIMARY KEY, title TEXT NOT NULL, state TEXT NOT NULL, completed INTEGER NOT NULL, total INTEGER NOT NULL, created_at TEXT NOT NULL, error TEXT);");

            var version = GetMeta("schema_version");
            var initialized = GetMeta("initialized") == "1";
            if (initialized && version is null)
                throw new InvalidDataException("Initialized SQLite database is missing schema_version.");
            if (version is not null && version != "1" && version != "2")
                throw new InvalidDataException("Unsupported SQLite schema " + version);

            // Idempotent v1 -> v2 migration. This also repairs a fresh database if
            // the process stopped after DDL but before the first data Commit.
            using (var tx = _connection.BeginTransaction())
            {
                EnsureColumn("jobs", "kind", "TEXT NOT NULL DEFAULT 'demo'", tx);
                EnsureColumn("jobs", "model", "TEXT", tx);
                EnsureColumn("jobs", "prompt", "TEXT", tx);
                EnsureColumn("jobs", "result", "TEXT", tx);
                SetMeta("schema_version", Schema.ToString(CultureInfo.InvariantCulture), tx);
                tx.Commit();
            }

            Exec(@"CREATE TABLE IF NOT EXISTS chat_messages (
                id TEXT PRIMARY KEY, session TEXT NOT NULL, role TEXT NOT NULL,
                text TEXT NOT NULL, model TEXT, created_at TEXT NOT NULL);");
            Exec("CREATE INDEX IF NOT EXISTS chat_messages_session_idx ON chat_messages (session, created_at);");
        }
        catch
        {
            _connection.Dispose();
            throw;
        }
    }

    private void Exec(string sql)
    {
        using var c = _connection.CreateCommand();
        c.CommandText = sql;
        c.ExecuteNonQuery();
    }

    private string? GetMeta(string key)
    {
        using var c = _connection.CreateCommand();
        c.CommandText = "SELECT value FROM metadata WHERE key=$key";
        c.Parameters.AddWithValue("$key", key);
        return c.ExecuteScalar() as string;
    }

    private void SetMeta(string key, string value, SqliteTransaction tx)
    {
        using var c = _connection.CreateCommand();
        c.Transaction = tx;
        c.CommandText = @"INSERT INTO metadata(key,value) VALUES ($key,$value)
            ON CONFLICT(key) DO UPDATE SET value=excluded.value";
        c.Parameters.AddWithValue("$key", key);
        c.Parameters.AddWithValue("$value", value);
        c.ExecuteNonQuery();
    }

    private void EnsureColumn(string table, string name, string definition, SqliteTransaction tx)
    {
        var exists = false;
        using (var info = _connection.CreateCommand())
        {
            info.Transaction = tx;
            info.CommandText = "PRAGMA table_info(" + table + ");";
            using var reader = info.ExecuteReader();
            while (reader.Read())
                if (string.Equals(reader.GetString(1), name, StringComparison.OrdinalIgnoreCase))
                    exists = true;
        }
        if (exists) return;
        using var alter = _connection.CreateCommand();
        alter.Transaction = tx;
        alter.CommandText = "ALTER TABLE " + table + " ADD COLUMN " + name + " " + definition;
        alter.ExecuteNonQuery();
    }

    private static DateTimeOffset Date(string? value, string field, string id)
    {
        if (!DateTimeOffset.TryParseExact(value, "O", CultureInfo.InvariantCulture,
            DateTimeStyles.RoundtripKind, out var result))
            throw new InvalidDataException($"Ungültiger SQLite-Zeitwert {field} bei {id}.");
        return result;
    }

    private static string DateText(DateTimeOffset value) => value.ToString("O", CultureInfo.InvariantCulture);

    public Snapshot? Read()
    {
        lock (_dbLock)
        {
            if (GetMeta("initialized") != "1") return null;
            if (GetMeta("schema_version") != Schema.ToString(CultureInfo.InvariantCulture))
                throw new InvalidDataException("Initialized SQLite database has missing or unsupported schema.");
            var s = new Snapshot { SchemaVersion = Schema };
            var updated = GetMeta("updated_at")
                ?? throw new InvalidDataException("SQLite updated_at metadata missing.");
            s.UpdatedAt = Date(updated, "updated_at", "metadata");
            using (var c = _connection.CreateCommand())
            {
                c.CommandText = "SELECT id,text,created_at FROM decisions ORDER BY created_at,id";
                using var r = c.ExecuteReader();
                while (r.Read())
                {
                    var id = r.GetString(0);
                    var t = r.GetString(1);
                    if (string.IsNullOrWhiteSpace(id) || string.IsNullOrWhiteSpace(t))
                        throw new InvalidDataException("Ungültiger Beschluss " + id);
                    s.Decisions.Add(new Decision(id, t, Date(r.GetString(2), "created_at", id)));
                }
            }
            using (var c = _connection.CreateCommand())
            {
                c.CommandText = "SELECT id,title,state,completed,total,created_at,error,kind,model,prompt,result FROM jobs ORDER BY created_at,id";
                using var r = c.ExecuteReader();
                while (r.Read())
                {
                    var id = r.GetString(0);
                    var completed = r.GetInt32(3);
                    var total = r.GetInt32(4);
                    if (string.IsNullOrWhiteSpace(id) || total < 1 || completed < 0 || completed > total)
                        throw new InvalidDataException("Ungültige Auftragsdaten " + id);
                    s.Jobs.Add(new Job(id, r.GetString(1), r.GetString(2), completed, total,
                        Date(r.GetString(5), "created_at", id),
                        r.IsDBNull(6) ? null : r.GetString(6),
                        r.GetString(7), r.IsDBNull(8) ? null : r.GetString(8),
                        r.IsDBNull(9) ? null : r.GetString(9), r.IsDBNull(10) ? null : r.GetString(10)));
                }
            }
            using (var c = _connection.CreateCommand())
            {
                c.CommandText = @"SELECT id,session,role,text,model,created_at FROM
                   (SELECT * FROM chat_messages WHERE session='main' ORDER BY created_at DESC,id DESC LIMIT 80)
                   ORDER BY created_at,id";
                using var r = c.ExecuteReader();
                while (r.Read())
                {
                    var id = r.GetString(0);
                    var role = r.GetString(2);
                    if (role is not ("user" or "assistant") || string.IsNullOrWhiteSpace(id))
                        throw new InvalidDataException("Ungültige Chat-Nachricht " + id);
                    s.ChatMessages.Add(new ChatMessage(id, r.GetString(1), role, r.GetString(3),
                        r.IsDBNull(4) ? null : r.GetString(4), Date(r.GetString(5), "created_at", id)));
                }
            }
            return s;
        }
    }

    public void Commit(Snapshot candidate)
    {
        lock (_dbLock)
        {
            if (candidate.SchemaVersion != Schema) throw new InvalidDataException("Unsupported Snapshot contract version.");
            using var tx = _connection.BeginTransaction();
            // No full-table delete: preserve historical rows and use targeted UPSERTs.
            foreach (var d in candidate.Decisions)
            {
                using var c = _connection.CreateCommand();
                c.Transaction = tx;
                c.CommandText = @"INSERT INTO decisions(id,text,created_at) VALUES ($id,$text,$date)
                    ON CONFLICT(id) DO UPDATE SET text=excluded.text";
                c.Parameters.AddWithValue("$id", d.Id);
                c.Parameters.AddWithValue("$text", d.Text);
                c.Parameters.AddWithValue("$date", DateText(d.CreatedAt));
                c.ExecuteNonQuery();
            }
            foreach (var j in candidate.Jobs) UpsertJob(j, tx);
            SetMeta("schema_version", Schema.ToString(CultureInfo.InvariantCulture), tx);
            SetMeta("updated_at", DateText(candidate.UpdatedAt), tx);
            SetMeta("initialized", "1", tx);
            tx.Commit();
        }
    }

    // Chat request and its user turn are one transaction: a DB failure cannot
    // leave an un-runnable queued job or an unmatched user message.
    public void CreateChatRequest(Job job, ChatMessage userMessage)
    {
        lock (_dbLock)
        {
            using var tx = _connection.BeginTransaction();
            UpsertJob(job, tx);
            InsertChatMessage(userMessage, tx);
            SetMeta("updated_at", DateText(userMessage.CreatedAt), tx);
            SetMeta("schema_version", Schema.ToString(CultureInfo.InvariantCulture), tx);
            SetMeta("initialized", "1", tx);
            tx.Commit();
        }
    }

    // Streaming writes touch only the active job row, not every decision/job row.
    public void UpdateJobProgress(string id, string state, int completed, int total,
        string? result, string? error, DateTimeOffset updatedAt)
    {
        lock (_dbLock)
        {
            using var tx = _connection.BeginTransaction();
            using var c = _connection.CreateCommand();
            c.Transaction = tx;
            c.CommandText = @"UPDATE jobs SET state=$state,completed=$completed,total=$total,
                result=$result,error=$error WHERE id=$id";
            c.Parameters.AddWithValue("$id", id);
            c.Parameters.AddWithValue("$state", state);
            c.Parameters.AddWithValue("$completed", completed);
            c.Parameters.AddWithValue("$total", total);
            c.Parameters.AddWithValue("$result", (object?)result ?? DBNull.Value);
            c.Parameters.AddWithValue("$error", (object?)error ?? DBNull.Value);
            if (c.ExecuteNonQuery() != 1) throw new InvalidDataException("Chat-Auftrag nicht gefunden: " + id);
            SetMeta("updated_at", DateText(updatedAt), tx);
            tx.Commit();
        }
    }

    // Final job state and the assistant completion/error/cancel marker commit together.
    public void FinishChat(string id, string state, string? result, string? error,
        ChatMessage assistantMessage, DateTimeOffset updatedAt)
    {
        lock (_dbLock)
        {
            using var tx = _connection.BeginTransaction();
            using (var c = _connection.CreateCommand())
            {
                c.Transaction = tx;
                c.CommandText = @"UPDATE jobs SET state=$state,completed=$completed,result=$result,error=$error WHERE id=$id";
                c.Parameters.AddWithValue("$id", id);
                c.Parameters.AddWithValue("$state", state);
                c.Parameters.AddWithValue("$completed", state == "done" ? 1 : 0);
                c.Parameters.AddWithValue("$result", (object?)result ?? DBNull.Value);
                c.Parameters.AddWithValue("$error", (object?)error ?? DBNull.Value);
                if (c.ExecuteNonQuery() != 1) throw new InvalidDataException("Chat-Auftrag nicht gefunden: " + id);
            }
            InsertChatMessage(assistantMessage, tx);
            SetMeta("updated_at", DateText(updatedAt), tx);
            tx.Commit();
        }
    }

    public void AppendChatMessage(ChatMessage message)
    {
        lock (_dbLock)
        {
            using var tx = _connection.BeginTransaction();
            InsertChatMessage(message, tx);
            SetMeta("updated_at", DateText(message.CreatedAt), tx);
            tx.Commit();
        }
    }

    private void InsertChatMessage(ChatMessage message, SqliteTransaction tx)
    {
        using var c = _connection.CreateCommand();
        c.Transaction = tx;
        c.CommandText = @"INSERT INTO chat_messages(id,session,role,text,model,created_at)
            VALUES ($id,$session,$role,$text,$model,$date)";
        c.Parameters.AddWithValue("$id", message.Id);
        c.Parameters.AddWithValue("$session", message.Session);
        c.Parameters.AddWithValue("$role", message.Role);
        c.Parameters.AddWithValue("$text", message.Text);
        c.Parameters.AddWithValue("$model", (object?)message.Model ?? DBNull.Value);
        c.Parameters.AddWithValue("$date", DateText(message.CreatedAt));
        c.ExecuteNonQuery();
    }

    private void UpsertJob(Job job, SqliteTransaction tx)
    {
        using var c = _connection.CreateCommand();
        c.Transaction = tx;
        c.CommandText = @"INSERT INTO jobs(id,title,state,completed,total,created_at,error,kind,model,prompt,result)
            VALUES ($id,$title,$state,$completed,$total,$date,$error,$kind,$model,$prompt,$result)
            ON CONFLICT(id) DO UPDATE SET state=excluded.state,completed=excluded.completed,
               total=excluded.total,error=excluded.error,result=excluded.result";
        c.Parameters.AddWithValue("$id", job.Id);
        c.Parameters.AddWithValue("$title", job.Title);
        c.Parameters.AddWithValue("$state", job.State);
        c.Parameters.AddWithValue("$completed", job.Completed);
        c.Parameters.AddWithValue("$total", job.Total);
        c.Parameters.AddWithValue("$date", DateText(job.CreatedAt));
        c.Parameters.AddWithValue("$error", (object?)job.Error ?? DBNull.Value);
        c.Parameters.AddWithValue("$kind", job.Kind);
        c.Parameters.AddWithValue("$model", (object?)job.Model ?? DBNull.Value);
        c.Parameters.AddWithValue("$prompt", (object?)job.Prompt ?? DBNull.Value);
        c.Parameters.AddWithValue("$result", (object?)job.Result ?? DBNull.Value);
        c.ExecuteNonQuery();
    }

    public void Dispose() => _connection.Dispose();
}

using System.Net.Http.Json;
using System.Text.Json;
using System.Runtime.CompilerServices;
using System.Text;

namespace Genesis.Ollama;

public sealed record ChatTurn(string Role, string Content);

public sealed class OllamaClient : IDisposable
{
    private readonly HttpClient _http;
    private readonly bool _ownsClient;
    private const int MaxResponseBytes = 2 * 1024 * 1024;

    public OllamaClient(HttpClient? http = null)
    {
        _ownsClient = http is null;
        _http = http ?? new HttpClient
        {
            BaseAddress = new Uri("http://127.0.0.1:11434/"),
            Timeout = TimeSpan.FromMinutes(3)
        };
    }

    public async Task<string[]> ModelsAsync(CancellationToken cancellation = default)
    {
        using var response = await _http.GetAsync("api/tags", HttpCompletionOption.ResponseHeadersRead, cancellation);
        response.EnsureSuccessStatusCode();
        using var json = await ReadBoundedJsonAsync(response.Content, cancellation);
        if (!json.RootElement.TryGetProperty("models", out var models) || models.ValueKind != JsonValueKind.Array)
            return [];
        return models.EnumerateArray().Where(x => x.TryGetProperty("name", out var name) && name.ValueKind == JsonValueKind.String)
            .Select(x => x.GetProperty("name").GetString()!)
            .Where(x => !string.IsNullOrWhiteSpace(x))
            .ToArray();
    }

    public async Task<string> ChatAsync(string model, string userText, CancellationToken cancellation = default)
    {
        if (string.IsNullOrWhiteSpace(model) || model.Length > 160)
            throw new ArgumentException("Modellname fehlt oder ist zu lang.", nameof(model));
        if (string.IsNullOrWhiteSpace(userText) || userText.Length > 16000)
            throw new ArgumentException("Nachricht muss 1 bis 16.000 Zeichen enthalten.", nameof(userText));
        using var request = new HttpRequestMessage(HttpMethod.Post, "api/chat")
        {
            Content = JsonContent.Create(new
            {
                model,
                stream = false,
                messages = new[] { new { role = "user", content = userText } },
                options = new { num_predict = 2048 }
            })
        };
        using var response = await _http.SendAsync(request, HttpCompletionOption.ResponseHeadersRead, cancellation);
        response.EnsureSuccessStatusCode();
        using var json = await ReadBoundedJsonAsync(response.Content, cancellation);
        if (!json.RootElement.TryGetProperty("message", out var message) ||
            !message.TryGetProperty("content", out var content) || content.ValueKind != JsonValueKind.String)
            throw new InvalidDataException("Ungültige Modellantwort.");
        return content.GetString() ?? "";
    }

    public async IAsyncEnumerable<string> ChatStreamAsync(
        string model, IReadOnlyList<ChatTurn> messages,
        [EnumeratorCancellation] CancellationToken cancellation = default)
    {
        if (string.IsNullOrWhiteSpace(model) || model.Length > 160)
            throw new ArgumentException("Modellname fehlt oder ist zu lang.", nameof(model));
        if (messages.Count == 0 || messages.Count > 40 ||
            messages.Any(m => m.Role is not ("user" or "assistant" or "system")
                || string.IsNullOrWhiteSpace(m.Content) || m.Content.Length > 16000))
            throw new ArgumentException("Ungültiger Chat-Kontext.", nameof(messages));
        using var request = new HttpRequestMessage(HttpMethod.Post, "api/chat")
        {
            Content = JsonContent.Create(new
            {
                model,
                stream = true,
                messages = messages.Select(m => new { role = m.Role, content = m.Content }).ToArray(),
                options = new { num_predict = 2048 }
            })
        };
        using var response = await _http.SendAsync(request, HttpCompletionOption.ResponseHeadersRead, cancellation);
        await CheckStatusAsync(response, cancellation);
        await using var stream = await response.Content.ReadAsStreamAsync(cancellation);
        using var reader = new StreamReader(stream, Encoding.UTF8, detectEncodingFromByteOrderMarks: false, leaveOpen: true);
        var total = 0;
        var done = false;
        while (!done)
        {
            var line = await ReadBoundedLineAsync(reader, 64 * 1024, cancellation);
            if (line is null) throw new InvalidDataException("Ollama stream ended without done=true.");
            if (string.IsNullOrWhiteSpace(line)) continue;
            using var json = JsonDocument.Parse(line);
            var root = json.RootElement;
            if (root.TryGetProperty("error", out var error))
                throw new InvalidDataException("Ollama: " + (error.GetString() ?? "Fehler"));
            if (root.TryGetProperty("message", out var message) &&
                message.TryGetProperty("content", out var content) &&
                content.ValueKind == JsonValueKind.String)
            {
                var chunk = content.GetString() ?? "";
                total += chunk.Length;
                if (total > 250000)
                    throw new InvalidDataException("Ollama output exceeded 250000 characters.");
                if (chunk.Length > 0) yield return chunk;
            }
            if (root.TryGetProperty("done", out var complete) && complete.ValueKind == JsonValueKind.True)
                done = true;
        }
    }

    private static async Task CheckStatusAsync(HttpResponseMessage response, CancellationToken cancellation)
    {
        if (response.IsSuccessStatusCode) return;
        string detail;
        try
        {
            using var json = await ReadBoundedJsonAsync(response.Content, cancellation);
            detail = json.RootElement.TryGetProperty("error", out var error) &&
                error.ValueKind == JsonValueKind.String ? error.GetString() ?? "Fehler" : "HTTP-Fehler";
        }
        catch (Exception ex) when (ex is JsonException or InvalidDataException)
        {
            detail = "HTTP-Fehler";
        }
        throw new HttpRequestException($"Ollama HTTP {(int)response.StatusCode}: {detail}",
            null, response.StatusCode);
    }

    private static async Task<string?> ReadBoundedLineAsync(StreamReader reader, int maxChars, CancellationToken ct)
    {
        var result = new StringBuilder();
        var one = new char[1];
        while (true)
        {
            var n = await reader.ReadAsync(one.AsMemory(), ct);
            if (n == 0) return result.Length == 0 ? null
                : throw new InvalidDataException("Incomplete Ollama stream message.");
            if (one[0] == '\n') return result.ToString().TrimEnd('\r');
            if (result.Length >= maxChars) throw new InvalidDataException("Ollama stream message too large.");
            result.Append(one[0]);
        }
    }

    private static async Task<JsonDocument> ReadBoundedJsonAsync(HttpContent content, CancellationToken cancellation)
    {
        if (content.Headers.ContentLength > MaxResponseBytes)
            throw new InvalidDataException("Ollama-Antwort überschreitet das Größenlimit.");

        await using var stream = await content.ReadAsStreamAsync(cancellation);
        using var buffer = new MemoryStream();
        var chunk = new byte[8192];
        while (true)
        {
            var count = await stream.ReadAsync(chunk.AsMemory(), cancellation);
            if (count == 0) break;
            if (buffer.Length + count > MaxResponseBytes)
                throw new InvalidDataException("Ollama-Antwort überschreitet das Größenlimit.");
            buffer.Write(chunk, 0, count);
        }
        buffer.Position = 0;
        return await JsonDocument.ParseAsync(buffer, cancellationToken: cancellation);
    }

    public void Dispose()
    {
        if (_ownsClient) _http.Dispose();
    }
}

            using var reader = new StreamReader(client, Encoding.UTF8, leaveOpen: true);
            await writer.WriteLineAsync(JsonSerializer.Serialize(new Request(cmd, text, id, authToken, model), Protocol.WireJson));
            using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(10));
            var line = await reader.ReadLineAsync(timeout.Token);
            return JsonSerializer.Deserialize<Response>(line ?? throw new Exception("Null response"), Protocol.WireJson)
                ?? throw new Exception("No response object");
        }
        catch (Exception ex) when (ex is IOException or TimeoutException)
        {
            last = ex;
            await Task.Delay(200);
        }
    }
    throw new Exception("Host IPC unreachable", last);
}
async Task<Response> RawAsync(string line)
{
    using var client = new NamedPipeClientStream(".", pipeName, PipeDirection.InOut, PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
    await client.ConnectAsync(2000);
    using var writer = new StreamWriter(client, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
    using var reader = new StreamReader(client, Encoding.UTF8, leaveOpen: true);
    await writer.WriteLineAsync(line);
    return JsonSerializer.Deserialize<Response>((await reader.ReadLineAsync())!, Protocol.WireJson)!;
}
async Task WaitForChatBodyAsync(string prompt)

    for (var i = 0; i < 100; i++)

            catch (JsonException) { }

    }
    throw new Exception("Fake Ollama did not receive prompt: " + prompt);
}

sealed class FakeOllama : IAsyncDisposable
{
    private readonly TcpListener _listener = new(IPAddress.Loopback, 0);
    private readonly CancellationTokenSource _stop = new();
    private readonly Task _acceptTask;
    public string Endpoint { get; }
    public volatile bool DelayChat;
    public volatile bool FailMidStream;
    private string? _lastChatBody;
    public string? LastChatBody => Volatile.Read(ref _lastChatBody);

    public FakeOllama()
    {
        _listener.Start();
        Endpoint = $"http://127.0.0.1:{((IPEndPoint)_listener.LocalEndpoint).Port}/";
        _acceptTask = Task.Run(AcceptAsync);
    }

    private async Task AcceptAsync()
    {
        while (!_stop.IsCancellationRequested)
        {
            try
            {
                var client = await _listener.AcceptTcpClientAsync(_stop.Token);
                _ = Task.Run(() => ReplyAsync(client));
            }
            catch (OperationCanceledException) { break; }
            catch (ObjectDisposedException) { break; }
        }
    }

    private async Task ReplyAsync(TcpClient client)
    {
        using (client)
        {
            try
            {
                var stream = client.GetStream();
                var header = new List<byte>();
                var byteBuffer = new byte[1];
                while (header.Count < 16384)
                {
                    if (await stream.ReadAsync(byteBuffer, _stop.Token) == 0) return;
                    header.Add(byteBuffer[0]);
                    var n = header.Count;
                    if (n >= 4 && header[n-4] == 13 && header[n-3] == 10 &&
                        header[n-2] == 13 && header[n-1] == 10) break;
                }
                var request = Encoding.ASCII.GetString(header.ToArray());
                // HttpClient may send JSON using either Content-Length or chunked framing.
                // The request body must be drained before closing, otherwise Windows
                // may reset the TCP connection while the client is reading the response.
                var headers = request.Split("\r\n", StringSplitOptions.RemoveEmptyEntries);
                var lengthLine = headers.FirstOrDefault(x => x.StartsWith("Content-Length:", StringComparison.OrdinalIgnoreCase));
                var isChunked = headers.Any(x => x.StartsWith("Transfer-Encoding:", StringComparison.OrdinalIgnoreCase)
                    && x.Contains("chunked", StringComparison.OrdinalIgnoreCase));
                using var requestBody = new MemoryStream();
                if (lengthLine is not null && int.TryParse(lengthLine[(lengthLine.IndexOf(':') + 1)..].Trim(), out var bodyLength))
                {
                    var bodyBytes = await ReadExactAsync(stream, bodyLength, _stop.Token);
                    requestBody.Write(bodyBytes);
                }
                else if (isChunked)
                {
                    while (true)
                    {
                        var lengthText = await ReadAsciiLineAsync(stream, _stop.Token);
                        var size = Convert.ToInt32(lengthText.Split(';')[0], 16);
                        if (size == 0)
                        {
                            while ((await ReadAsciiLineAsync(stream, _stop.Token)).Length > 0) { }
                            break;
                        }
                        var chunk = await ReadExactAsync(stream, size, _stop.Token);
                        requestBody.Write(chunk);
                        await ReadExactAsync(stream, 2, _stop.Token); // CRLF
                    }
                }
                var requestBodyText = Encoding.UTF8.GetString(requestBody.ToArray());
                string body;
                if (request.StartsWith("GET /api/tags", StringComparison.Ordinal))
                {
                    body = "{\"models\":[{\"name\":\"fake-alpha:1b\"}]}";
                }
                else if (request.StartsWith("POST /api/chat", StringComparison.Ordinal))
                {
                    var delayThisChat = DelayChat;
                    var failThisChat = FailMidStream;
                    Volatile.Write(ref _lastChatBody, requestBodyText);
                    if (delayThisChat) await Task.Delay(5000, _stop.Token);
                    body = failThisChat
                        ? "{\"message\":{\"role\":\"assistant\",\"content\":\"Teilantwort\"},\"done\":false}\n" +
                          "{\"error\":\"synthetic mid-stream failure\"}\n"
                        : "{\"message\":{\"role\":\"assistant\",\"content\":\"Hallo von \"},\"done\":false}\n" +
                          "{\"message\":{\"role\":\"assistant\",\"content\":\"Fake-Ollama!\"},\"done\":true}\n";
                }
                else
                {
                    body = "{}";
                }
                var bytes = Encoding.UTF8.GetBytes(body);
                var head = Encoding.ASCII.GetBytes("HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nContent-Length: " +
                    bytes.Length + "\r\nConnection: close\r\n\r\n");
                await stream.WriteAsync(head, _stop.Token);
                await stream.WriteAsync(bytes, _stop.Token);
                await stream.FlushAsync(_stop.Token);
            }
            catch (Exception ex) when (ex is IOException or SocketException or OperationCanceledException) { }
        }
    }

    private static async Task<byte[]> ReadExactAsync(NetworkStream stream, int count, CancellationToken cancellation)
    {
        if (count < 0 || count > 2 * 1024 * 1024) throw new InvalidDataException("Invalid mock body length.");
        var result = new byte[count];
        var offset = 0;
        while (offset < count)
        {
            var read = await stream.ReadAsync(result.AsMemory(offset, count - offset), cancellation);
            if (read == 0) throw new EndOfStreamException();
            offset += read;
        }
        return result;
    }

    private static async Task<string> ReadAsciiLineAsync(NetworkStream stream, CancellationToken cancellation)
    {
        var bytes = new List<byte>();
        var one = new byte[1];
        while (bytes.Count < 1024)
        {
            if (await stream.ReadAsync(one, cancellation) == 0) throw new EndOfStreamException();
            if (one[0] == 10) return Encoding.ASCII.GetString(bytes.ToArray()).TrimEnd('\r');
            bytes.Add(one[0]);
        }
        throw new InvalidDataException("Mock HTTP header line too large.");
    }

    public async ValueTask DisposeAsync()
    {
        _stop.Cancel();
        _listener.Stop();
        try { await _acceptTask; } catch (OperationCanceledException) { }
        _stop.Dispose();
    }
}

# GENESIS 0.3.1-R1 — Windows Alpha-Abnahme (Entwurf)

**Kein Produktions-Release.** Diese Testfassung ist ein Review-/Alpha-Kandidat. Eine kontrollierte lokale Alpha ist erst nach erfolgreichem Windows-CI-Lauf und manueller Abnahme auf dem Zielrechner bestätigt.

## Was R1 gegenüber 0.3.1 korrigiert

- Snapshot-Vertragsversion und SQLite-Schema sind beide explizit Version 2; der Legacy-JSON-Importer akzeptiert alte Version-1-Snapshots und normalisiert sie vor dem SQLite-Commit.
- SQLite-v1→v2-Migration ist idempotent: fehlende Spalten werden über `PRAGMA table_info` erkannt, das Schema-Metadatum wird in derselben DDL-Transaktion auf v2 gesetzt. Ein Neustart nach bereits ausgeführter DDL kann die Migration fortsetzen.
- Chat-Auftrag und User-Nachricht werden atomar in einer SQLite-Transaktion angelegt. Ein Fehler beim Einfügen der Nachricht hinterlässt keinen verwaisten `queued`-Job.
- Chat-Abbruch verwendet den Zwischenstatus `canceling`. Erst nach Ende des Ollama-Streams und Freigabe des ModelGate wird der Job terminal als `canceled` markiert.
- Erfolgreiche, abgebrochene und fehlgeschlagene Generierungen schreiben jeweils eine Assistant-Abschluss-/Statusnachricht in derselben Transaktion wie den terminalen Jobstatus. Ältere unbeantwortete User-Turns werden beim Kontextaufbau herausgefiltert.
- Streaming-Zwischenstände aktualisieren nur den betreffenden SQLite-Jobdatensatz; Beschlüsse und andere Jobs werden nicht alle 400 ms erneut geschrieben.
- SQLite-Korruption über `SqliteException` löst ebenfalls eine Quarantäne-Kopie aus. Datenbank und vorhandene WAL-/SHM-Sidecars werden erhalten; Quarantäne ist keine automatische Reparatur.
- Der Chat-Kontext ist begrenzt: maximal 5.000 Zeichen Beschlusskontext, 3.000 Zeichen abgeschlossene Gesprächshistorie und 8.000 Zeichen aktuelle Eingabe. Unbeantwortete User-Nachrichten werden nicht als Historie weitergereicht.
- Das Command Deck zeigt Fehler des letzten Chat-Auftrags, zeigt `canceling` verständlich an und überschreibt den Transcript nicht beim Modellladen.
- Beim Aktualisieren des Transcripts werden Scrollposition und Textauswahl soweit möglich bewahrt. Das Chatpanel ist per GridSplitter in der Höhe veränderbar.
- Buildversion: `0.3.1.1`, InformationalVersion `0.3.1-R1-alpha`. CI-Artefakt: `GENESIS-0.3.1-R1-win-x64-UNREVIEWED`.

## Start der Testfassung

1. Windows 11 x64 und .NET 8 Desktop Runtime installieren. Ollama lokal starten und ein passendes Modell vorhalten (beispielsweise `qwen3.5:9b`).
2. GitHub Actions im privaten Entwicklungsrepo: nur **erfolgreichen** Lauf für den exakten R1-PR-Commit verwenden. Artefakt `GENESIS-0.3.1-R1-win-x64-UNREVIEWED` herunterladen und vollständig entpacken.
3. Im entpackten Stammordner `Genesis.Launcher.exe` doppelklicken. **Nicht** nur die EXE ohne die `UI/`- und `Host/`-Unterordner kopieren.
4. Command Deck: `Modelle laden`, Modell auswählen, Nachricht eingeben, Enter senden. Status/Teilantwort und Abbrechen prüfen.
5. Eine zweite Nachricht senden und kontrollieren, ob die vorherige vollständige User-/Assistant-Runde im Kontext berücksichtigt wird. Host beenden, GUI erneut öffnen und Verlauf prüfen.
6. Während einer langen Antwort auf `Abbrechen` klicken. Der Status soll zunächst `canceling` anzeigen. Eine zweite Modellgenerierung darf erst starten, wenn die erste Modellressource tatsächlich freigegeben wurde.
7. Ollama bewusst beenden oder ein nicht vorhandenes Modell wählen. Die konkrete Fehlermeldung muss im Transcript erscheinen.
8. Light/Dark und Fenstergröße auf den beiden PC-Monitoren sowie dem 4K-TV bei unterschiedlichen DPI-Skalierungen überprüfen. Chatpanel mit dem Trenner vergrößern/verkleinern; während Streaming nach oben scrollen und Text markieren. Screenshots und Auffälligkeiten notieren.
9. Host während eines laufenden Chats kontrolliert beenden und anschließend den Verlauf nach Neustart prüfen.

## Schutz bestehender Daten

Der Host verwendet standardmäßig `%LOCALAPPDATA%\JoshuaSpike5Genesis\genesis.db`. Vor dem ersten R1-Test mit wertvollen GENESIS-Daten diesen **gesamten Ordner bei beendetem Host** sichern, einschließlich `genesis.db-wal` und `genesis.db-shm`, falls vorhanden.

R1 verwendet weiterhin SQLite-Schema v2. Ein 0.3.0-Host ist nicht kompatibel mit SQLite-Schema v2. **Kein automatischer Downgrade.** Snapshot-Vertragsversion 2 ist von der physischen SQLite-Schemaversion zu unterscheiden, auch wenn beide hier bewusst denselben Wert tragen.

Alte `state.json`-Quelldateien werden beim Erstimport nicht gelöscht. Ungültige SQLite-Werte oder eine erkannte SQLite-Korruption führen zu einer Quarantäne-Kopie statt stiller Neuinitialisierung. Quarantäne ersetzt kein Backup.

## Automatisch geprüfte Kriterien

- Host/UI/Launcher kompilieren und R1-Paket wird vollständig veröffentlicht
- Protokoll-Roundtrip und Ollama-API-Vertrag
- Legacy JSON→SQLite und SQLite-v1→v2 Migration
- Wiederanlauf einer bereits teilweise ausgeführten frischen Migration
- Atomare Chat-Erstellung: doppelter Chat-Message-Key darf keinen verwaisten Job hinterlassen
- Beschluss- und vorherige User-/Assistant-Kontextturns werden im Fake-Ollama-Request geprüft
- Ollama-Fehler mitten im Stream werden als fehlgeschlagener Job plus Assistant-Statusnachricht gespeichert
- Abbruch bleibt während einer simulierten Modellfreigabe sichtbar; parallele Generierung wird abgewiesen
- Host-Stopp während eines aktiven Chats wartet auf den Worker und persistiert den Abbruch
- SQLite-Wertefehler und nicht lesbare SQLite-Datei werden quarantänisiert
- Verlauf bleibt nach Host-Neustart erhalten

**Noch ausstehend:** realer Ollama-/Qwen-Durchlauf, visuelle Interaktion mit WPF, GPU-Auslastungs-/12-GB-VRAM-Messung, 4K-TV-/Misch-DPI-Abnahme und mehrstündiger Dauerbetrieb auf dem Windows-Rechner. GitHub Actions testet keinen echten lokalen Qwen-Prozess und ersetzt keinen manuellen UI-Test.
