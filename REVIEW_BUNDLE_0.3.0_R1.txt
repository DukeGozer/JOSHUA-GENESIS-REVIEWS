using System.IO.Pipes;
using System.Text;
using System.Text.Json;
using Genesis.Contracts;
using Genesis.Storage;
using Genesis.Ollama;

namespace Genesis.Host;

internal static class Program
{
    private static readonly object StateGate = new();
    private static readonly object LogGate = new();
    private static readonly CancellationTokenSource StopSignal = new();
    private static readonly string DataDirectory = (Protocol.TestOverridesEnabled
        ? Environment.GetEnvironmentVariable("GENESIS_DATA_DIR") : null)
        ?? Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "JoshuaSpike5Genesis");
    private static readonly string StatePath = Path.Combine(DataDirectory, "state.json");
    private static readonly string BackupPath = StatePath + ".bak";
    private static readonly string LogPath = Path.Combine(DataDirectory, "host.log");
    private static readonly string ControlTokenPath = Path.Combine(DataDirectory, "control.token");
    private static string _controlToken = "";
    private static Snapshot _state = new();
    private static SnapshotStore? _store;
    private static readonly OllamaClient LocalModel = new();
    private static string _startupNotice = "GENESIS Host erreichbar";
    private const int MaxRequestChars = 64 * 1024;
    private const int MaxStoredEntries = 500;
    private const long MaxLogBytes = 1024 * 1024;

    private static async Task<int> Main()
    {
        Directory.CreateDirectory(DataDirectory);
        try
        {
            using var singleton = new FileStream(Path.Combine(DataDirectory, "host.lock"),
                FileMode.OpenOrCreate, FileAccess.ReadWrite, FileShare.None);
            LoadFromStorage();
            // Capability token prevents accidental or blind stop requests, not malicious same-user processes.
            _controlToken = Convert.ToHexString(System.Security.Cryptography.RandomNumberGenerator.GetBytes(32));
            File.WriteAllText(ControlTokenPath, _controlToken, new UTF8Encoding(false));
            Log("GENESIS 0.1.2 Host started.");
            var worker = Task.Run(() => WorkLoop(StopSignal.Token));
            var clientSlots = new SemaphoreSlim(16, 16);
            var activeClients = new List<Task>();
            try
            {
                while (!StopSignal.IsCancellationRequested)
                {
                    await clientSlots.WaitAsync(StopSignal.Token).ConfigureAwait(false);
                    NamedPipeServerStream? server = null;
                    try
                    {
                        server = new NamedPipeServerStream(Protocol.ActivePipeName,
                            PipeDirection.InOut, 16, PipeTransmissionMode.Byte,
                            PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
                        await server.WaitForConnectionAsync(StopSignal.Token);
                        // The task owns the connected pipe and releases its concurrency slot.
                        var connected = server;
                        server = null;
                        var task = ProcessClientAsync(connected, StopSignal.Token, clientSlots);
                        activeClients.Add(task);
                        activeClients.RemoveAll(t => t.IsCompleted);
                    }
                    catch (OperationCanceledException) when (StopSignal.IsCancellationRequested)
                    {
                        server?.Dispose();
                        clientSlots.Release();
                        break;
                    }
                    catch (Exception ex)
                    {
                        server?.Dispose();
                        clientSlots.Release();
                        Log("Pipe accept error: " + ex);
                        await Task.Delay(200, StopSignal.Token);
                    }
                }
            }
            catch (OperationCanceledException) when (StopSignal.IsCancellationRequested) { }
            finally
            {
                StopSignal.Cancel();
                try { await Task.WhenAll(activeClients); }
                catch (Exception ex) { Log("Client shutdown error: " + ex); }
                try { await worker; } catch (OperationCanceledException) { }
                clientSlots.Dispose();
                _store?.Dispose();
                try { File.Delete(ControlTokenPath); } catch (IOException ex) { Log("Could not delete control token: " + ex.Message); }
                Log("GENESIS host stopped.");
            }
            return 0;
        }
        catch (IOException ex) { Log("Host lock unavailable (possibly already running): " + ex.Message); return 2; }
        catch (Exception ex) { Log("Host startup failed: " + ex); return 1; }
    }

    private static async Task ProcessClientAsync(
        NamedPipeServerStream pipe, CancellationToken stopping, SemaphoreSlim slots)
    {
        try
        {
            using (pipe) await HandleClient(pipe, stopping);
        }
        catch (Exception ex) { Log("Unhandled client error: " + ex); }
        finally { slots.Release(); }
    }

    private static async Task HandleClient(NamedPipeServerStream pipe, CancellationToken stopping)
    {
        // Handle one request per connection; every read has a deadline and size bound.
        using var reader = new StreamReader(pipe, Encoding.UTF8, detectEncodingFromByteOrderMarks: false, leaveOpen: true);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        try
        {
            using var timeout = CancellationTokenSource.CreateLinkedTokenSource(stopping);
            timeout.CancelAfter(TimeSpan.FromSeconds(10));
            var line = await PipeLineCodec.ReadBoundedAsync(reader, MaxRequestChars, timeout.Token);
            if (line is null) return;
            Request? request;
            try { request = JsonSerializer.Deserialize<Request>(line, Protocol.WireJson); }
            catch (JsonException)
            {
                await writer.WriteLineAsync(JsonSerializer.Serialize(
                    new Response(false, "Ungültiges JSON."), Protocol.WireJson));
                return;
            }

            // Ollama may take minutes. Never hold the state lock or block other pipe clients.
            if (request?.Command is not null &&
                (request.Command.Equals("models", StringComparison.OrdinalIgnoreCase) ||
                 request.Command.Equals("chat", StringComparison.OrdinalIgnoreCase)))
            {
                using var modelTimeout = CancellationTokenSource.CreateLinkedTokenSource(stopping);
                modelTimeout.CancelAfter(TimeSpan.FromMinutes(3));
                var modelResponse = await HandleModelRequestAsync(request, modelTimeout.Token);
                await writer.WriteLineAsync(JsonSerializer.Serialize(modelResponse, Protocol.WireJson));
                return;
            }

            Response response;
            lock (StateGate)
            {
                try { response = Execute(request); }
                catch (IOException ex) { Log("Persistence I/O error: " + ex); response = new(false, "Speichern fehlgeschlagen. Keine Änderung bestätigt."); }
                catch (UnauthorizedAccessException ex) { Log("Persistence permission error: " + ex); response = new(false, "Speichern nicht erlaubt."); }
            }
            await writer.WriteLineAsync(JsonSerializer.Serialize(response, Protocol.WireJson));
            if (request?.Command?.Equals("stop", StringComparison.OrdinalIgnoreCase) == true && response.Success)
                StopSignal.Cancel();
        }
        catch (OperationCanceledException) when (!stopping.IsCancellationRequested)
        {
            await TryReplyAsync(writer, new(false, "Zeitüberschreitung beim Lesen."));
        }
        catch (InvalidDataException) { await TryReplyAsync(writer, new(false, "Anfrage zu groß.")); }
        catch (OperationCanceledException) when (stopping.IsCancellationRequested) { }
        catch (Exception ex) { Log("Client error: " + ex); await TryReplyAsync(writer, new(false, "Anfrage fehlgeschlagen.")); }
    }

    private static async Task<Response> HandleModelRequestAsync(Request request, CancellationToken ct)
    {
        try
        {
            if (request.Command.Equals("models", StringComparison.OrdinalIgnoreCase))
            {
                var models = await LocalModel.ModelsAsync(ct);
                return new(true, models.Length == 0 ? "Ollama erreichbar, keine Modelle installiert." : "Ollama-Modelle geladen.",
                    Models: models);
            }
            if (request.Text is null || request.Model is null)
                return new(false, "Nachricht und Modellname erforderlich.");
            if (request.Text.Length > 16000 || request.Model.Length > 160)
                return new(false, "Nachricht oder Modellname zu lang.");
            var answer = await LocalModel.ChatAsync(request.Model, request.Text, ct);
            if (answer.Length > 250000) answer = answer[..250000] + "\n[Ausgabe gekürzt]";
            return new(true, "Antwort von lokalem Ollama.", ChatText: answer);
        }
        catch (HttpRequestException ex) { Log("Ollama HTTP error: " + ex); return new(false, "Ollama nicht erreichbar oder HTTP-Fehler. Läuft Ollama lokal?"); }
        catch (OperationCanceledException) { return new(false, "Ollama-Zeitlimit überschritten."); }
        catch (Exception ex) { Log("Ollama error: " + ex); return new(false, "Modellanfrage fehlgeschlagen. Details in host.log."); }
    }

    private static async Task TryReplyAsync(StreamWriter writer, Response response)
    {
        try { await writer.WriteLineAsync(JsonSerializer.Serialize(response, Protocol.WireJson)); }
        catch (IOException) { }
        catch (ObjectDisposedException) { }
    }

    private static Response Execute(Request? request)
    {
        if (string.IsNullOrWhiteSpace(request?.Command)) return new(false, "Befehl fehlt.");
        switch (request.Command.Trim().ToLowerInvariant())
        {
            case "status": return new(true, _startupNotice, Clone());
            case "remember":
                if (string.IsNullOrWhiteSpace(request.Text)) return new(false, "Bitte Beschlusstext angeben.");
                if (request.Text.Length > 2000) return new(false, "Beschlusstext zu lang (max. 2000 Zeichen).");
                if (_state.Decisions.Count >= MaxStoredEntries) return new(false, "Beschlusslimit erreicht (500).");
                var decided = Clone();
                decided.Decisions.Add(new Decision(Guid.NewGuid().ToString("N"), request.Text.Trim(), DateTimeOffset.UtcNow));
                Save(decided);
                return new(true, "Beschluss gespeichert.", Clone());
            case "start-demo":
                if (_state.Jobs.Count >= MaxStoredEntries) return new(false, "Auftragslimit erreicht (500).");
                var started = Clone();
                started.Jobs.Add(new Job(Guid.NewGuid().ToString("N"), "SPIKE-Testauftrag: 10 Schritte",
                    "queued", 0, 10, DateTimeOffset.UtcNow));
                Save(started);
                return new(true, "Testauftrag gestartet.", Clone());
            case "cancel":
                if (string.IsNullOrWhiteSpace(request.Id)) return new(false, "Auftrags-ID fehlt.");
                var index = _state.Jobs.FindIndex(x => x.Id == request.Id);
                if (index < 0) return new(false, "Auftrag nicht gefunden.");
                if (_state.Jobs[index].State is "done" or "canceled") return new(false, "Auftrag bereits beendet.");
                var cancelled = Clone();
                cancelled.Jobs[index] = cancelled.Jobs[index] with { State = "canceled" };
                Save(cancelled);
                return new(true, "Auftrag abgebrochen.", Clone());
            case "stop":
                if (request.AuthToken is null || request.AuthToken.Length != _controlToken.Length ||
                    !System.Security.Cryptography.CryptographicOperations.FixedTimeEquals(
                        Encoding.ASCII.GetBytes(request.AuthToken), Encoding.ASCII.GetBytes(_controlToken)))
                    return new(false, "Host-Stopp nicht autorisiert.");
                return new(true, "Host wird kontrolliert beendet.");
            default: return new(false, "Unbekannter Befehl.");
        }
    }

    private static async Task WorkLoop(CancellationToken stopping)
    {
        while (!stopping.IsCancellationRequested)
        {
            try
            {
                await Task.Delay(TimeSpan.FromSeconds(1), stopping);
                lock (StateGate)
                {
                    var index = _state.Jobs.FindIndex(x => x.State is "queued" or "running");
                    if (index >= 0)
                    {
                        var next = Clone();
                        var job = next.Jobs[index];
                        var completed = Math.Min(job.Total, job.Completed + 1);
                        next.Jobs[index] = job with
                        {
                            State = completed == job.Total ? "done" : "running",
                            Completed = completed
                        };
                        Save(next);
                    }
                }
            }
            catch (OperationCanceledException) when (stopping.IsCancellationRequested) { break; }
            catch (Exception ex)
            {
                Log("Background job paused after error: " + ex);
                try { await Task.Delay(TimeSpan.FromSeconds(5), stopping); }
                catch (OperationCanceledException) { break; }
            }
        }
    }

    private static Snapshot Clone() => new()
    {
        SchemaVersion = _state.SchemaVersion,
        Decisions = [.. _state.Decisions],
        Jobs = [.. _state.Jobs],
        UpdatedAt = _state.UpdatedAt
    };

    private static Snapshot ReadSnapshot(string path)
    {
        var result = JsonSerializer.Deserialize<Snapshot>(File.ReadAllText(path), Protocol.Json)
            ?? throw new InvalidDataException("Snapshot is empty.");
        if (result.SchemaVersion != 1) throw new InvalidDataException("Unknown state schema " + result.SchemaVersion);
        if (result.Decisions is null || result.Jobs is null) throw new InvalidDataException("Missing state collections.");
        return result;
    }

    private static void LoadFromStorage()
    {
        _store = new SnapshotStore(DataDirectory);
        var existing = _store.Read();
        if (existing is not null)
        {
            _state = existing;
            _startupNotice = "GENESIS SQLite-Zustand geladen.";
            return;
        }

        // Import is retried if the process stops before SQLite marks itself initialized.
        // Original legacy JSON and backups are NEVER deleted or overwritten by migration.
        Load();
        _store.Commit(_state);
        _startupNotice = File.Exists(StatePath)
            ? "JSON-Zustand in SQLite migriert; Originaldateien erhalten."
            : "Neue SQLite-Datenbank initialisiert.";
    }

    private static void Load()
    {
        if (!File.Exists(StatePath))
        {
            if (File.Exists(BackupPath))
            {
                // Refuse silent loss of a previously saved snapshot.
                _state = ReadSnapshot(BackupPath);
                File.Copy(BackupPath, StatePath, overwrite: false);
                _startupNotice = "Zustand aus Backup wiederhergestellt (Hauptdatei fehlte).";
            }
            return;
        }
        try { _state = ReadSnapshot(StatePath); }
        catch (Exception ex) when (ex is IOException or JsonException or InvalidDataException)
        {
            Log("Primary state invalid: " + ex);
            if (!File.Exists(BackupPath)) throw new InvalidDataException("No backup; original preserved.", ex);
            var backup = ReadSnapshot(BackupPath); // verify backup before touching primary
            var quarantined = StatePath + ".corrupt-" + DateTimeOffset.UtcNow.ToString("yyyyMMddHHmmssfff");
            File.Move(StatePath, quarantined);
            File.Copy(BackupPath, StatePath);
            _state = backup;
            _startupNotice = "Beschädigter Zustand gesichert; Backup wiederhergestellt.";
            Log(_startupNotice + " " + quarantined);
        }
    }

    // SQLite transaction is authoritative. Publish in-memory state only after COMMIT.
    private static void Save(Snapshot candidate)
    {
        candidate.UpdatedAt = DateTimeOffset.UtcNow;
        (_store ?? throw new InvalidOperationException("SQLite is not initialized.")).Commit(candidate);
        _state = candidate;
    }

    private static void Log(string message)
    {
        try
        {
            lock (LogGate)
            {
                if (File.Exists(LogPath) && new FileInfo(LogPath).Length >= MaxLogBytes)
                    File.Move(LogPath, LogPath + ".old", overwrite: true);
                File.AppendAllText(LogPath, $"{DateTimeOffset.UtcNow:u} {message}{Environment.NewLine}");
            }
        }
        catch { /* Logging must never terminate the host. */ }
    }
}

using System.IO;

namespace Genesis.Host;

public static class PipeLineCodec
{
    // A char-by-char reader bounds memory even if a client never sends a newline.
    public static async Task<string?> ReadBoundedAsync(StreamReader reader, int limit, CancellationToken ct)
    {
        if (limit < 1) throw new ArgumentOutOfRangeException(nameof(limit));
        var line = new System.Text.StringBuilder();
        var chars = new char[1];
        while (true)
        {
            var read = await reader.ReadAsync(chars.AsMemory(), ct);
            if (read == 0) return line.Length == 0 ? null : throw new InvalidDataException("Incomplete request.");
            if (chars[0] == '\n') return line.ToString().TrimEnd('\r');
            if (line.Length >= limit) throw new InvalidDataException("Request exceeds max length.");
            line.Append(chars[0]);
        }
    }
}

    private static async Task<Response> SendAsync(Request request)
    {
        using var pipe = new NamedPipeClientStream(".", Protocol.ActivePipeName, PipeDirection.InOut,
            PipeOptions.Asynchronous | PipeOptions.CurrentUserOnly);
        await pipe.ConnectAsync(1500);
        using var writer = new StreamWriter(pipe, new UTF8Encoding(false), leaveOpen: true) { AutoFlush = true };
        using var reader = new StreamReader(pipe, Encoding.UTF8, leaveOpen: true);
        await writer.WriteLineAsync(JsonSerializer.Serialize(request, Protocol.WireJson));
        using var timeout = new CancellationTokenSource(request.Command == "chat" ? TimeSpan.FromMinutes(3) : TimeSpan.FromSeconds(10));
        var reply = await ReadBoundedReplyAsync(reader, 4 * 1024 * 1024, timeout.Token);
        return reply is null ? new(false, "Keine Antwort vom Host")
            : JsonSerializer.Deserialize<Response>(reply, Protocol.WireJson)
              ?? new(false, "Ungültige Antwort");
    }

    private static async Task<string?> ReadBoundedReplyAsync(StreamReader reader, int limit, CancellationToken cancellation)
    {
        var result = new StringBuilder();
        var buffer = new char[1];
        while (true)
        {
            var count = await reader.ReadAsync(buffer.AsMemory(), cancellation);
            if (count == 0) return result.Length == 0 ? null : throw new InvalidDataException("Unvollständige Antwort.");
            if (buffer[0] == '\n') return result.ToString().TrimEnd('\r');
            if (result.Length >= limit) throw new InvalidDataException("Host-Antwort überschreitet das Größenlimit.");
            result.Append(buffer[0]);
        }
    }

using System.Globalization;
using Genesis.Contracts;
using Microsoft.Data.Sqlite;

namespace Genesis.Storage;

/// <summary>
/// SQLite is the authoritative GENESIS 0.2 state. All decisions/jobs are committed
/// in a single transaction; interrupted transactions never expose partial snapshots.
/// </summary>
public sealed class SnapshotStore : IDisposable
{
    private readonly SqliteConnection _connection;

    public SnapshotStore(string dataDirectory)
    {
        Directory.CreateDirectory(dataDirectory);
        var path = Path.Combine(dataDirectory, "genesis.db");
        _connection = new SqliteConnection(new SqliteConnectionStringBuilder
        {
            DataSource = path,
            Mode = SqliteOpenMode.ReadWriteCreate,
            Cache = SqliteCacheMode.Private,
            Pooling = false
        }.ToString());
        _connection.Open();
        Exec("PRAGMA journal_mode=WAL;");
        Exec("PRAGMA busy_timeout=5000;");
        Exec("PRAGMA synchronous=FULL;");
        Exec("CREATE TABLE IF NOT EXISTS metadata (key TEXT PRIMARY KEY, value TEXT NOT NULL);");
        Exec("CREATE TABLE IF NOT EXISTS decisions (id TEXT PRIMARY KEY, text TEXT NOT NULL, created_at TEXT NOT NULL);");
        Exec("CREATE TABLE IF NOT EXISTS jobs (id TEXT PRIMARY KEY, title TEXT NOT NULL, state TEXT NOT NULL, completed INTEGER NOT NULL, total INTEGER NOT NULL, created_at TEXT NOT NULL, error TEXT);");
        var version = GetMeta("schema_version");
        if (version is not null && version != "1")
            throw new InvalidDataException("Unsupported SQLite schema " + version);
    }

    private void Exec(string sql)
    {
        using var c = _connection.CreateCommand();
        c.CommandText = sql;
        c.ExecuteNonQuery();
    }

    private string? GetMeta(string key)
    {
        using var c = _connection.CreateCommand();
        c.CommandText = "SELECT value FROM metadata WHERE key=$key";
        c.Parameters.AddWithValue("$key", key);
        return c.ExecuteScalar() as string;
    }

    public Snapshot? Read()
    {
        if (GetMeta("initialized") != "1") return null;
        if (GetMeta("schema_version") != "1")
            throw new InvalidDataException("Initialized SQLite database has missing or unsupported schema.");
        var s = new Snapshot { SchemaVersion = 1 };
        var updated = GetMeta("updated_at");
        if (updated is not null) s.UpdatedAt = DateTimeOffset.Parse(updated, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind);
        using (var c = _connection.CreateCommand())
        {
            c.CommandText = "SELECT id,text,created_at FROM decisions ORDER BY created_at,id";
            using var r = c.ExecuteReader();
            while (r.Read())
                s.Decisions.Add(new Decision(r.GetString(0), r.GetString(1),
                    DateTimeOffset.Parse(r.GetString(2), CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind)));
        }
        using (var c = _connection.CreateCommand())
        {
            c.CommandText = "SELECT id,title,state,completed,total,created_at,error FROM jobs ORDER BY created_at,id";
            using var r = c.ExecuteReader();
            while (r.Read())
                s.Jobs.Add(new Job(r.GetString(0), r.GetString(1), r.GetString(2),
                    r.GetInt32(3), r.GetInt32(4),
                    DateTimeOffset.Parse(r.GetString(5), CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind),
                    r.IsDBNull(6) ? null : r.GetString(6)));
        }
        return s;
    }

    public void Commit(Snapshot candidate)
    {
        if (candidate.SchemaVersion != 1) throw new InvalidDataException("Unsupported Snapshot version.");
        using var tx = _connection.BeginTransaction();
        using (var clean = _connection.CreateCommand())
        {
            clean.Transaction = tx;
            clean.CommandText = "DELETE FROM decisions; DELETE FROM jobs;";
            clean.ExecuteNonQuery();
        }
        foreach (var d in candidate.Decisions)
        {
            using var c = _connection.CreateCommand();
            c.Transaction = tx;
            c.CommandText = "INSERT INTO decisions (id,text,created_at) VALUES ($id,$text,$date)";
            c.Parameters.AddWithValue("$id", d.Id);
            c.Parameters.AddWithValue("$text", d.Text);
            c.Parameters.AddWithValue("$date", d.CreatedAt.ToString("O", CultureInfo.InvariantCulture));
            c.ExecuteNonQuery();
        }
        foreach (var j in candidate.Jobs)
        {
            using var c = _connection.CreateCommand();
            c.Transaction = tx;
            c.CommandText = @"INSERT INTO jobs (id,title,state,completed,total,created_at,error)
                VALUES ($id,$title,$state,$completed,$total,$date,$error)";
            c.Parameters.AddWithValue("$id", j.Id);
            c.Parameters.AddWithValue("$title", j.Title);
            c.Parameters.AddWithValue("$state", j.State);
            c.Parameters.AddWithValue("$completed", j.Completed);
            c.Parameters.AddWithValue("$total", j.Total);
            c.Parameters.AddWithValue("$date", j.CreatedAt.ToString("O", CultureInfo.InvariantCulture));
            c.Parameters.AddWithValue("$error", (object?)j.Error ?? DBNull.Value);
            c.ExecuteNonQuery();
        }
        foreach (var (key,value) in new[] { ("schema_version", "1"), ("updated_at", candidate.UpdatedAt.ToString("O", CultureInfo.InvariantCulture)), ("initialized", "1") })
        {
            using var c = _connection.CreateCommand();
            c.Transaction = tx;
            c.CommandText = "INSERT INTO metadata(key,value) VALUES ($key,$value) ON CONFLICT(key) DO UPDATE SET value=excluded.value";
            c.Parameters.AddWithValue("$key", key);
            c.Parameters.AddWithValue("$value", value);
            c.ExecuteNonQuery();
        }
        tx.Commit();
    }

    public void Dispose() => _connection.Dispose();
}

            throw new ArgumentException("Modellname fehlt oder ist zu lang.", nameof(model));
        if (string.IsNullOrWhiteSpace(userText) || userText.Length > 16000)
            throw new ArgumentException("Nachricht muss 1 bis 16.000 Zeichen enthalten.", nameof(userText));
        using var request = new HttpRequestMessage(HttpMethod.Post, "api/chat")
        {
            Content = JsonContent.Create(new
            {
                model,
                stream = false,
                messages = new[] { new { role = "user", content = userText } },
                options = new { num_predict = 2048 }
            })
        };
        using var response = await _http.SendAsync(request, HttpCompletionOption.ResponseHeadersRead, cancellation);
        response.EnsureSuccessStatusCode();
        using var json = await ReadBoundedJsonAsync(response.Content, cancellation);
        if (!json.RootElement.TryGetProperty("message", out var message) ||
            !message.TryGetProperty("content", out var content) || content.ValueKind != JsonValueKind.String)
            throw new InvalidDataException("Ungültige Modellantwort.");
        return content.GetString() ?? "";
    }

    private static async Task<JsonDocument> ReadBoundedJsonAsync(HttpContent content, CancellationToken cancellation)
    {
        if (content.Headers.ContentLength > MaxResponseBytes)
            throw new InvalidDataException("Ollama-Antwort überschreitet das Größenlimit.");

        await using var stream = await content.ReadAsStreamAsync(cancellation);
        using var buffer = new MemoryStream();
        var chunk = new byte[8192];
        while (true)
        {
            var count = await stream.ReadAsync(chunk.AsMemory(), cancellation);
            if (count == 0) break;
            if (buffer.Length + count > MaxResponseBytes)
                throw new InvalidDataException("Ollama-Antwort überschreitet das Größenlimit.");
            buffer.Write(chunk, 0, count);
        }
        buffer.Position = 0;
        return await JsonDocument.ParseAsync(buffer, cancellationToken: cancellation);
    }

    public void Dispose()
    {
        if (_ownsClient) _http.Dispose();
    }
